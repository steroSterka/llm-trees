def predict_heart_disease(X: dict):
    """
    Predicts whether a patient has heart disease based on their features.
    
    Parameters:
    X (dict): A dictionary containing the patient's features.
        - 'age': age in years
        - 'sex': sex (male, female)
        - 'chest_pain': Chest Pain Type (3 = Typical angina, 0 = Asymptomatic, 2 = Non-anginal pain, 1 = Atypical angina)
        - 'trestbps': Resting Blood Pressure in mmHg
        - 'chol': Serum Cholesterol in mg/dl
        - 'fbs': Fasting Blood Sugar (1 = false, 2 = true)
        - 'restecg': Resting Electrocardiographic Results (2 = Normal, 3 = Having ST-T wave abnormality, 1 = Showing probable or definite left ventricular hypertrophy by Estes' criteria)
        - 'thalach': Maximum Heart Rate Achieved in bpm
        - 'exang': Exercise Induced Angina (1 = no, 2 = yes)
        - 'oldpeak': ST Depression Induced by Exercise Relative to Rest (0 to 5)
        - 'slope': Slope of the Peak Exercise ST Segment (3 = Upsloping, 2 = Flat, 1 = Downsloping)
        - 'ca': Number of Major Vessels Colored by Fluoroscopy (0, 1, 2, 3)
        - 'thal': Thalassemia (2 = Normal, 1 = Fixed defect, 3 = Reversible defect)

    Returns:
    tuple: A tuple containing the predicted target variable and a list of boolean values representing the truth values of the inner nodes.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = [None, None]

    # Node 1: Check if Chest Pain Type is Typical angina (3)
    nodes[0] = X['chest_pain'] == 3

    # If Chest Pain Type is Typical angina, check Maximum Heart Rate Achieved
    if nodes[0]:
        # Node 2: Check if Maximum Heart Rate Achieved is less than or equal to 120 bpm
        nodes[1] = X['thalach'] <= 120
        # Prediction based on the decision tree
        prediction = int(nodes[1])
    else:
        # If Chest Pain Type is not Typical angina, consider other factors
        # For simplicity, let's check if Maximum Heart Rate Achieved is less than or equal to 100 bpm
        nodes[1] = X['thalach'] <= 100
        # Prediction based on the decision tree
        prediction = int(not nodes[1])

    return prediction, [int(node) for node in nodes]
