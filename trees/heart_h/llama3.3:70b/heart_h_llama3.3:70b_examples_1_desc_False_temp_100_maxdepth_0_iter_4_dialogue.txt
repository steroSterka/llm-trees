def predict_heart_disease(X: dict):
    """
    Predicts whether a patient has heart disease or not based on their characteristics.

    Parameters:
    X (dict): A dictionary containing patient characteristics.
        - age (int): Age in years.
        - sex (str): Sex of the patient ('male' or 'female').
        - chest_pain (int): Type of chest pain (0 = Asymptomatic, 1 = Atypical angina, 2 = Non-anginal pain, 3 = Typical angina).
        - trestbps (int): Resting blood pressure in mmHg.
        - chol (int): Serum cholesterol level in mg/dl.
        - fbs (int): Fasting blood sugar level (1 = false, 2 = true).
        - restecg (int): Resting electrocardiographic results (0 = Normal, 1 = Showing probable or definite left ventricular hypertrophy by Estes' criteria, 2 = Having ST-T wave abnormality).
        - thalach (int): Maximum heart rate achieved in bpm.
        - exang (int): Exercise induced angina (1 = no, 2 = yes).
        - oldpeak (float): ST depression induced by exercise relative to rest.
        - slope (int): Slope of the peak exercise ST segment (1 = Downsloping, 2 = Flat, 3 = Upsloping).
        - ca (int): Number of major vessels colored by fluoroscopy (0, 1, 2, 3).
        - thal (int): Thalassemia (1 = Fixed defect, 2 = Normal, 3 = Reversible defect).

    Returns:
    prediction (int): Predicted target variable (0 = No heart disease, 1 = Heart disease).
    nodes (list): List of boolean values representing the conditions in the decision tree.
    """
    nodes = [None] * 7

    # Node 0: Age > 40
    nodes[0] = X['age'] > 40

    # Node 1: Chest pain type is Typical angina (3)
    if nodes[0]:
        nodes[1] = X['chest_pain'] == 3
    else:
        nodes[1] = False

    # Node 2: Resting blood pressure > 120 mmHg
    if not nodes[1]:
        nodes[2] = X['trestbps'] > 120
    else:
        nodes[2] = False

    # Node 3: Fasting blood sugar is true (2)
    if nodes[2]:
        nodes[3] = X['fbs'] == 2
    else:
        nodes[3] = False

    # Node 4: Serum cholesterol > 200 mg/dl
    if not nodes[1] and not nodes[2]:
        nodes[4] = X['chol'] > 200
    else:
        nodes[4] = False

    # Node 5: Exercise induced angina is yes (2)
    if nodes[3] or nodes[4]:
        nodes[5] = X['exang'] == 2
    else:
        nodes[5] = False

    # Node 6: ST depression > 1
    if not nodes[5]:
        nodes[6] = X['oldpeak'] > 1
    else:
        nodes[6] = False

    prediction = 0  # Default to no heart disease

    if nodes[1]:  # Typical angina, likely heart disease
        prediction = 1
    elif nodes[5]:  # Exercise induced angina and other conditions, likely heart disease
        prediction = 1
    elif nodes[6] and (nodes[3] or nodes[4]):  # ST depression and high cholesterol or fasting blood sugar, possible heart disease
        prediction = 1

    return prediction, [int(node) for node in nodes]
