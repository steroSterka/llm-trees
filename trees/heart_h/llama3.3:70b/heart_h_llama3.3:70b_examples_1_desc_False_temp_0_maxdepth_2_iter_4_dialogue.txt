def predict(X: dict):
    """
    Predicts whether a patient has heart disease or not based on their features.
    
    Parameters:
    X (dict): A dictionary containing the patient's features.
        - age (int): Age in years
        - sex (str): Sex (male, female)
        - chest_pain (int): Chest Pain Type (3 = Typical angina, 0 = Asymptomatic, 2 = Non-anginal pain, 1 = Atypical angina)

        - thalach (int): Maximum Heart Rate Achieved in bpm
        - exang (int): Exercise Induced Angina (1 = no, 2 = yes)
        - oldpeak (float): ST Depression Induced by Exercise Relative to Rest (0 to 5)
        - slope (int): Slope of the Peak Exercise ST Segment (3 = Upsloping, 2 = Flat, 1 = Downsloping)
        - ca (int): Number of Major Vessels Colored by Fluoroscopy (0, 1, 2, 3)
        - thal (int): Thalassemia (2 = Normal, 1 = Fixed defect, 3 = Reversible defect)
    
    Returns:
    prediction (int): Whether the patient has heart disease or not (1 = yes, 0 = no)
    nodes (list): A list of boolean values representing the truth values of the inner nodes
    """
    
    # Initialize the nodes list with None values
    nodes = 2 * [None]
    
    # Node 0: Chest Pain Type is Typical angina or Atypical angina
    nodes[0] = X["chest_pain"] in [1, 3]
    
    # Node 1: Age is greater than 50 and Serum Cholesterol is greater than 200 mg/dl
    nodes[1] = X["age"] > 50 and X["chol"] > 200
    
    # Make the prediction based on the nodes
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Patient has heart disease
        else:

    else:

            prediction = 1  # Patient has heart disease
        else:

    # Convert the nodes list to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
