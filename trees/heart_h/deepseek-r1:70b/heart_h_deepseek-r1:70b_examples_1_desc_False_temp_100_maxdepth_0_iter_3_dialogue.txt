def predict(X: dict):

        # If exercise-induced angina is present, check ST depression level
        if nodes[1]:
            # Node 2: ST Depression Induced by Exercise Relative to Rest is at least 1
            nodes[2] = X['oldpeak'] >= 1

            if nodes[2]:
                # Node 3: Number of Major Vessels Colored by Fluoroscopy greater than 0
                nodes[3] = X['ca'] > 0
                
                # Predict heart disease based on vessel condition
                prediction = 1 if nodes[3] else 0
            else:
                # If ST depression does not meet the threshold, check ECG results
                # Node 4: Resting Electrocardiographic Results are abnormal (not equal to 'Normal')
                nodes[4] = X['restecg'] != 2
                
                prediction = 1 if nodes[4] else 0
        else:
            # If no exercise-induced angina, predict based on ECG results
            nodes[2] = False  # Not applicable; set to False for consistency
            nodes[3] = False  # Not applicable; set to False for consistency
            nodes[4] = X['restecg'] != 2
            prediction = 1 if nodes[4] else 0
    else:
        # If age is 40 or younger, predict based on chest pain type and other factors
        nodes[1] = False  # Not applicable; set to False for consistency
        nodes[2] = False  # Not applicable; set to False for consistency
        nodes[3] = False  # Not applicable; set to False for consistency
        prediction = 0  # Default prediction for younger patients without meeting other conditions

    # Ensure all nodes are either True (1) or False (0)

    return prediction, nodes
