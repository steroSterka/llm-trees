def predict(X: dict):
    """
    Predicts which of the two boxing fighters won each round.

    Parameters:

        - 'Judge': Judge (0 = E. Williams, 1 = L. O'Connell, 2 = S. Christodoulu, 3 = HBO-Lederman, 4 = Sportsticker, 5 = Boxing Times, 6 = Sportsline, 7 = Associated Press, 8 = ESPN, 9 = Boxing Monthly-Leach)

    nodes (list): A list representing the truth values of the inner nodes.
    """

    # Initialize the nodes with None
    nodes = 2 * [None]

    # Check if Judge is HBO-Lederman (3) or not
    nodes[0] = X['Judge'] == 3

    # If it's not HBO-Lederman, check if Official judge is Yes (1)
    nodes[1] = X['Official'] == 1

    # Make the prediction based on the nodes
    if nodes[0]:
        prediction = 1  # Lewis wins if Judge is HBO-Lederman
    else:
        if nodes[1]:
            prediction = 0  # Holyfield wins if Official judge is Yes and not HBO-Lederman
        else:
            prediction = 1  # Lewis wins otherwise

    # Convert boolean values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
