def predict(X: dict):
    """
    Predicts which boxer won the round based on the judge, official status, and round number.
    
    Parameters:
    X (dict): Dictionary containing the features 'Judge', 'Official', 'Round'.
    
    Returns:
    prediction (int): The predicted winner of the round (1 = Lewis, 0 = Holyfield).
    nodes (list): A list representing the truth values of the inner nodes.
    """

    # Initialize the nodes list with None
    nodes = 2 * [None]

    # Node 1: Check if the judge is known to favor Lewis (e.g., HBO-Lederman, ESPN)
    nodes[0] = X["Judge"] in [3, 8]  # 3 = HBO-Lederman, 8 = ESPN

    # Node 2: Check if the round is early (Round <= 6)
    nodes[1] = X["Round"] <= 6

    # Make the prediction based on the nodes
    if nodes[0]:
        prediction = 1  # Lewis is favored by this judge
    else:
        if nodes[1]:
            prediction = 1  # Early round, Lewis might have an advantage
        else:
            prediction = 0  # Late round, Holyfield might gain momentum

    # Convert boolean nodes to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
