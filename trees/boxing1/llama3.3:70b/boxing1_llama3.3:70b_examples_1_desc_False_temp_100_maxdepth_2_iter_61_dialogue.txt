def predict(X: dict):

    Predicts which boxer won each round based on the given features.

    Args:

        - 'Judge': The judge scoring the round (0 = E. Williams, 1 = L. O'Connell, 2 = S. Christodoulu, 3 = HBO-Lederman, 4 = Sportsticker, 5 = Boxing Times, 6 = Sportsline, 7 = Associated Press, 8 = ESPN, 9 = Boxing Monthly-Leach)

        - 'Round': The round number (1 to 12)

    Returns:

    nodes (list): A list of truth values for the inner nodes
    """

    nodes = 2 * [None]

    # Check the judge for rounds where Lewis might be favored
    nodes[1] = (X['Judge'] in [3, 8]) if nodes[0] else (X['Judge'] not in [7, 9])

    # Make the prediction based on the node values
    if nodes[0]:
        if X['Judge'] in [3, 8]:  # HBO-Lederman or ESPN favor Lewis
            prediction = 1

    else:
