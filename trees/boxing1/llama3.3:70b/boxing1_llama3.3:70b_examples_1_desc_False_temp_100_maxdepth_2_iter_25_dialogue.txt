def predict(X: dict):
    """
    Predicts which boxer wins the round based on a decision tree.
    
    Parameters:
    X (dict): Dictionary containing features 'Judge', 'Official', 'Round', and target variable encoding.
    
    Returns:
    prediction (int): 1 if Lewis wins, 0 if Holyfield wins
    nodes (list): List of boolean values representing the conditions at each inner node
    """

    nodes = 2 * [None]

    # Node 0 condition: Judge is HBO-Lederman or Boxing Times (values 3 and 5)
    nodes[0] = X["Judge"] in [3, 5]
    
    # Node 1 condition: Round is greater than 6
    nodes[1] = X["Round"] > 6
    
    # Prediction based on the decision tree
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Lewis wins in later rounds with specific judges
        else:
            prediction = 0  # Holyfield wins in earlier rounds with specific judges
    else:
        if X["Official"] == 1:  # Official judge
            prediction = 1  # Lewis wins when it's an official judge and not a specific judge
        else:
            prediction = 0  # Holyfield wins otherwise

    return prediction, [int(node) for node in nodes]
