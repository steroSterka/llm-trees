def predict(X: dict):

    Predicts the winner of a boxing round (Lewis or Holyfield) based on a decision tree.

    Args:

                   as defined in the problem description.

    Returns:

               - The predicted winner (1 = Lewis, 0 = Holyfield).
               - A list representing the truth values of the inner nodes (1 = condition satisfied, 0 = condition not satisfied).

    # Node Truth Values list.  Will be populated as we traverse the tree
    node_values = []

    # --- Decision Tree Logic ---
    # Root Node: Round <= 4

    return prediction, node_values
