def predict(X: dict):
    # Initialize nodes with None
    nodes = [None, None]
    
    # First split: Official judge is yes (1)
    nodes[0] = X['Official'] == 1
    
    if nodes[0]:
        # Second split: Round <=6
        nodes[1] = X['Round'] <= 6
    else:
        # Second split: Judge is E. Williams or L. O'Connell (0,1)
        nodes[1] = X['Judge'] in [0, 1]
    
    # Determine prediction based on the nodes
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Lewis wins
        else:
            prediction = 0  # Holyfield wins
    else:
        if nodes[1]:
            prediction = 1  # Lewis wins
        else:
            prediction = 0  # Holyfield wins
    
    return prediction, nodes
