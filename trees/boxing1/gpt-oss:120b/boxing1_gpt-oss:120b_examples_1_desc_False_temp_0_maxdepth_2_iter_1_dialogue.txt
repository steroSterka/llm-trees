  * Any other judge → the round is predicted for **Lewis**.

    Parameters
    ----------

    Returns

        evaluated.  ``1`` means the condition is satisfied, ``0`` otherwise.
    """

    # Inner node 0 : is the round early (<= 6) or late (> 6) ?
    # ------------------------------------------------------------------
    node0 = X["Round"] <= 6          # True (1) → early round, False (0) → late round

    nodes = [int(node0), None]

    # ------------------------------------------------------------------

        node1 = X["Official"] == 1   # 1 = Yes, 0 = No

    return prediction, nodes
