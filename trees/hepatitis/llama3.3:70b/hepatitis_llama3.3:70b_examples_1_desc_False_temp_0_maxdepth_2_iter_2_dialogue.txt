def predict(X: dict):

    Predicts whether a patient survives or dies based on their medical features.
    
    Parameters:

              The keys should match the feature names exactly, including case and spelling.
    
    Returns:
    prediction (int): 0 if the patient is predicted to die, 1 if they are predicted to survive.
    nodes (list): A list of boolean values representing the truth values of the inner nodes in the decision tree.
                  The first element corresponds to the condition "ALBUMIN <= 2.5", and the second element corresponds to "PROTIME > 20" or "BILIRUBIN > 5".
    """
    
    # Initialize the list to store the node conditions
    nodes = 2 * [None]
    
    # Check if albumin level is less than or equal to 2.5

        nodes[1] = X["PROTIME"] > 20
    else:

            prediction = 0  # Die: low albumin and prolonged prothrombin time

        if nodes[1]:

    return prediction, [int(node) for node in nodes]
