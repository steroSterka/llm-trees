def predict_survival(X: dict):
    """
    Predicts whether a patient will survive or die based on their medical features.
    
    Parameters:
    X (dict): A dictionary containing the patient's medical features.
    
    Returns:
    prediction (int): 1 if the patient is predicted to survive, 0 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes in the decision tree.
    """
    
    # Initialize the nodes list with None values
    nodes = 5 * [None]

    nodes[0] = X["BILIRUBIN"] > 10
    
    # Node 1: Prothrombin time > 20 seconds
    nodes[1] = X["PROTIME"] > 20
    
    # Node 2: Albumin level < 2.5 g/dL
    nodes[2] = X["ALBUMIN"] < 2.5
    
    # Node 3: Spleen palpable
    nodes[3] = X["SPLEEN PALPABLE"] == 1
    
    # Node 4: Prothrombin time > 15 seconds (additional condition for patients with bilirubin level <= 10 mg/dL)
    nodes[4] = X["PROTIME"] > 15
    
    # Make predictions based on the node values
    if nodes[0]:
        prediction = 0  # Patient is predicted to die if bilirubin level is high
    elif nodes[1]:
        prediction = 0  # Patient is predicted to die if prothrombin time is long
    elif nodes[2]:
