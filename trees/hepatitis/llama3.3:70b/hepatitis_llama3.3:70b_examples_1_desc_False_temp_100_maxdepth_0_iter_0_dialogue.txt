def predict(X: dict):
    """
    Predicts patient survival based on given medical features.

    Parameters:
    X (dict): Dictionary containing the patient's medical features.
             Keys: 'AGE', 'SEX', 'STEROID', 'ANTIVIRALS', 'FATIGUE', 
                   'MALAISE', 'ANOREXIA', 'LIVER BIG', 'LIVER FIRM',

    Returns:
    prediction (int): 1 if the patient is predicted to survive, 0 otherwise.
    nodes (list): List of boolean values representing the conditions at each inner node.
    """

    # Initialize list to store the truth values of the inner nodes
    nodes = [None] * 4

    # Node 0: Bilirubin level <= 2.0 mg/dL
    nodes[0] = X['BILIRUBIN'] <= 2.0
    
    # If bilirubin level is low, check prothrombin time
    if nodes[0]:
        # Node 1: Prothrombin time <= 15 seconds
        nodes[1] = X['PROTIME'] <= 15
        
        # If both conditions are met, patient is likely to survive
        if nodes[1]:
            prediction = 1
            nodes[2:] = [False, False]
        else:
            # Node 2: Albumin level >= 3.5 g/dL (considering the patient might still survive with slightly higher prothrombin time)
            nodes[2] = X['ALBUMIN'] >= 3.5
            
            # If albumin level is adequate, consider survival
            if nodes[2]:
                prediction = 1
                nodes[3] = False
            else:
                prediction = 0
                nodes[3] = False
                
    # If bilirubin level is high, check albumin level directly for a chance of survival
    else:
        # Node 3: Albumin level >= 3.5 g/dL (direct check for survival chance with higher bilirubin)
        nodes[1:] = [False, X['ALBUMIN'] >= 3.5, False]
        
        if nodes[2]:
            prediction = 1
        else:
            prediction = 0

    # Convert boolean values to integers (1 for True, 0 for False) in the nodes list
    nodes = [int(node) for node in nodes]

    return prediction, nodes
