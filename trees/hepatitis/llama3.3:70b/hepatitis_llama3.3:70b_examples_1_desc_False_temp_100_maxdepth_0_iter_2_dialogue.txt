def predict(X: dict):
    """
    Decision Tree Classifier to predict patient survival.

    Parameters:
    X (dict): Dictionary containing patient features.
        - AGE (int): Age in years
        - SEX (int): Sex (2 = male, 1 = female)
        - STEROID (int): Steroids (0 = no, 1 = yes)
        - ANTIVIRALS (int): Antivirals (0 = no, 1 = yes)
        - FATIGUE (int): Fatigue (0 = no, 1 = yes)
        - MALAISE (int): Malaise (0 = no, 1 = yes)
        - ANOREXIA (int): Anorexia (0 = no, 1 = yes)
        - LIVER BIG (int): Liver big (0 = no, 1 = yes)
        - LIVER FIRM (int): Liver firm (0 = no, 1 = yes)
        - SPLEEN PALPABLE (int): Spleen palpable (0 = no, 1 = yes)
        - SPIDERS (int): Spider angiomata (0 = no, 1 = yes)
        - ASCITES (int): Ascites (0 = no, 1 = yes)
        - VARICES (int): Varices (0 = no, 1 = yes)
        - BILIRUBIN (float): Bilirubin level in mg/dL
        - ALK PHOSPHATE (float): Alkaline phosphatase level in units/liter
        - SGOT (float): Serum glutamic-oxaloacetic transaminase level in units/liter
        - ALBUMIN (float): Albumin level in g/dL
        - PROTIME (float): Prothrombin time in seconds
        - HISTOLOGY (int): Histologic data (0 = no, 1 = yes)

    Returns:
    prediction (int): Patient survival prediction (die = 0, live = 1)
    nodes (list): List of boolean values representing inner node conditions
    """

    # Initialize nodes list with None values
    nodes = 3 * [None]

    # Define conditions for each inner node
    nodes[0] = X["BILIRUBIN"] <= 2.0
    nodes[1] = X["PROTIME"] <= 15
    nodes[2] = X["ALBUMIN"] >= 3.5

    # Implement decision tree logic
    if nodes[0]:
        prediction = 1  # Patient survives if bilirubin level is low
    else:
        if nodes[1]:
            if nodes[2]:
                prediction = 1  # Patient survives if prothrombin time and albumin levels are within normal limits
            else:
                prediction = 0  # Patient dies if albumin level is low despite normal prothrombin time
        else:
            prediction = 0  # Patient dies if prothrombin time is prolonged

    return int(prediction), [int(node) for node in nodes]
