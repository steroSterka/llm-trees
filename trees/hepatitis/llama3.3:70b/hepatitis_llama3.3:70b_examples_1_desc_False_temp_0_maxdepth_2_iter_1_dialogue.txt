def predict(X: dict):
    """
    Decision Tree Classifier to Predict Patient Survival.

    Parameters:
    X (dict): Dictionary containing patient features.
    
    Returns:
    prediction (int): 0 if the patient dies, 1 if the patient survives.
    nodes (list): List of boolean values representing the truth values of inner nodes.
    """

    # Initialize nodes list with None
    nodes = 2 * [None]

    # Node 0: Bilirubin level <= 2.5 mg/dL
    nodes[0] = X["BILIRUBIN"] <= 2.5

    # If bilirubin is within the threshold, check albumin levels
    if nodes[0]:
        # Node 1: Albumin level > 3.0 g/dL
        nodes[1] = X["ALBUMIN"] > 3.0
        
        # Prediction based on albumin level
        if nodes[1]:
            prediction = 1  # Survives
        else:
            prediction = 0  # Dies
    else:
        # If bilirubin is above the threshold, check for ascites presence
        nodes[1] = X["ASCITES"] == 0
        
        # Prediction based on ascites presence
        if nodes[1]:
            prediction = 1  # Survives
        else:
            prediction = 0  # Dies

    # Convert boolean values in nodes to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
