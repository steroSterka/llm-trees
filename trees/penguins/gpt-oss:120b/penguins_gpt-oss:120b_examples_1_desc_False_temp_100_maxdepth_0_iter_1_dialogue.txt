def predict_penguin(X: dict):
    """
    Predict the penguin species (0 = Adelie, 1 = Chinstrap, 2 = Gentoo)
    from the feature dictionary X and return the truth‑values of the
    inner‑node tests.

    Expected keys in X (exact spelling):
        'island'               – 0 = Biscoe, 1 = Dream, 2 = Torgersen
        'culmen_length_mm'     – float / int
        'culmen_depth_mm'      – (unused in this tree)
        'flipper_length_mm'    – float / int
        'body_mass_g'          – float / int
        'sex'                  – 0 = nan, 1 = female, 2 = male

    Returns
    -------
    prediction : int
        0 → Adelie, 1 → Chinstrap, 2 → Gentoo
    nodes : list[int]
        List of 0/1 values, one for each inner node (in the order shown
        below). 1 means the condition evaluated to **True**, 0 means **False**.
    """

    # ------------------------------------------------------------------
    # 1️⃣  Inner‑node evaluations (computed *regardless* of the path taken)
    # ------------------------------------------------------------------
    # node 0 : is the island Biscoe ?
    node0 = X["island"] == 0
    # node 1 : is the island Dream ?
    node1 = X["island"] == 1
    # node 2 : (Biscoe branch) flipper length ≤ 215 mm ?
    node2 = X["flipper_length_mm"] <= 215
    # node 3 : (Torgersen branch) body mass ≤ 3800 g ?
    node3 = X["body_mass_g"] <= 3800
    # node 4 : (heavy Torgersen branch) culmen length ≤ 48 mm ?
    node4 = X["culmen_length_mm"] <= 48

    # store the boolean results as 0/1 as required by the specification
    nodes = [int(node0), int(node1), int(node2), int(node3), int(node4)]

    # ------------------------------------------------------------------
    # 2️⃣  Decision logic (mirrors the tree drawn in the previous answer)
    # ------------------------------------------------------------------
    if node0:                                 # island == Biscoe
        if node2:                             # flipper ≤ 215
            prediction = 0                    # Adelie
        else:
            prediction = 2                    # Gentoo
    elif node1:                               # island == Dream
        prediction = 1                        # Chinstrap
    else:                                     # island == Torgersen
        if node3:                             # body mass ≤ 3800
            prediction = 0                    # Adelie
        else:
            if node4:                         # culmen length ≤ 48
                prediction = 2                # Gentoo
            else:
                # The original tree added a sex check here, but both
                # possible outcomes are Gentoo, so we can return Gentoo
                # directly.
                prediction = 2                # Gentoo

    return prediction, nodes

### How the function works
| Index | Condition (inner node)                              | Meaning in the tree |
|------|------------------------------------------------------|---------------------|
| 0    | `X["island"] == 0`                                   | “island = Biscoe?”  |

The function always evaluates **all five** inner‑node conditions, stores their truth values (`1` for *True*, `0` for *False*), and then follows the appropriate branch to produce a single species prediction.
