```python
def predict(X: dict):
    # Initialize nodes with None (will be replaced by boolean values)
    nodes = [None] * 1  # One node for 'island' check
    
    if X['island'] == 0:
        prediction = 0  # Adelie
        nodes[0] = True
    elif X['island'] == 1:
        prediction = 1  # Chinstrap
        nodes[0] = True
    else:  # X['island'] == 2

        nodes[0] = True
    
    return prediction, [int(node) for node in nodes]
```
