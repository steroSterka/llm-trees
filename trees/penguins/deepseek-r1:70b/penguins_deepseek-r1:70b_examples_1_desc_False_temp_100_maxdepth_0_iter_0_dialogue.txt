def predict_penguin_species(X: dict):
    """
    Predicts the species of a penguin based on its characteristics using a decision tree.

    Args:
        X (dict): A dictionary containing the features:
            - 'island': The island where the penguin was recorded.
            - 'culmen_length_mm': Length of the culmen in millimeters.
            - 'culmen_depth_mm': Depth of the culmen in millimeters.
            - 'flipper_length_mm': Length of the flipper in millimeters.
            - 'body_mass_g': Body mass of the penguin in grams.
            - 'sex': Sex of the penguin (encoded as 0 for nan, 1 for female, 2 for male).

    Returns:
        tuple: A tuple containing:
            - prediction (int): The predicted species (0 = Adelie, 1 = Chinstrap, 2 = Gentoo)
            - nodes (list): A list of boolean values indicating whether each node's condition was met (1 if true, 0 otherwise).
    """

    # Initialize the nodes list with None to represent the structure of the decision tree
    nodes = [None] * 5

    # Node 0: Check if island is Biscoe
    nodes[0] = X['island'] == 'Biscoe'
    
    if nodes[0]:
        # Node 1: Flipper length > 200 mm for Gentoo
        nodes[1] = X['flipper_length_mm'] > 200
        if nodes[1]:
            prediction = 2  # Gentoo
        else:
            prediction = 0  # Adelie
    elif X['island'] == 'Dream':
        # Node 2: Culmen depth > 20 mm for Adelie
        nodes[2] = X['culmen_depth_mm'] > 20
        if nodes[2]:
            prediction = 0  # Adelie
        else:
            prediction = 1  # Chinstrap
    elif X['island'] == 'Torgersen':
        # Node 3: Body mass > 4000 grams for Gentoo
        nodes[3] = X['body_mass_g'] > 4000
        if nodes[3]:
            prediction = 2  # Gentoo

    nodes = [int(node) for node in nodes if node is not None]

    return prediction, nodes
