def predict_penguin_species(X: dict):
    """
    Predicts the penguin species (Adelie, Chinstrap, or Gentoo) based on given characteristics.
    
    Parameters:
    X (dict): Dictionary containing the features of a penguin. The keys should match the following:
        - 'island'
        - 'culmen_length_mm'
        - 'culmen_depth_mm'
        - 'flipper_length_mm'
        - 'body_mass_g'
        - 'sex'
    
    Returns:
    prediction (int): Predicted species of the penguin (0 = Adelie, 1 = Chinstrap, 2 = Gentoo)
    nodes (list): List representing the truth values of the inner nodes in the decision tree.
    """

    # Initialize nodes for tracking conditions
    nodes = 2 * [None]
    
    # Condition 1: Flipper length <= 190 mm to distinguish between Gentoo and other species
    nodes[0] = X['flipper_length_mm'] <= 190
    
    # If the first condition is met, use culmen depth to differentiate between Adelie and Chinstrap
    if nodes[0]:
        # Condition 2: Culmen depth <= 18.5 mm for Adelie vs. Chinstrap differentiation
        nodes[1] = X['culmen_depth_mm'] <= 18.5
        
        # Prediction based on conditions
        if nodes[1]:
            prediction = 0  # Adelie
        else:
            prediction = 1  # Chinstrap
    else:
        # If flipper length is more than 190 mm, predict Gentoo
        nodes[1] = False  # Not used in this branch but set for consistency
        prediction = 2  # Gentoo
    
    # Convert boolean values to integers (True -> 1, False -> 0)
    nodes = [int(node) if node is not None else None for node in nodes]
    
    return prediction, nodes

# Example usage:
example_penguin = {
    'island': 0,
    'culmen_length_mm': 45,

This function first checks the flipper length to decide if the penguin is likely a Gentoo or not. The `nodes` list keeps track of whether each condition in the decision tree was met, converted into integers for simplicity.
