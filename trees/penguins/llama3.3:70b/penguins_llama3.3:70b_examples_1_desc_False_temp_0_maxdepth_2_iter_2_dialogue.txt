def predict_penguin_species(X: dict):
    """
    Predicts the species of a penguin based on its characteristics.
    
    Parameters:
    X (dict): A dictionary containing the features of the penguin.
             - island: The island where the penguin was found (0 = Biscoe, 1 = Dream, 2 = Torgersen)
             - culmen_length_mm: The length of the penguin's culmen in mm
             - culmen_depth_mm: The depth of the penguin's culmen in mm
             - flipper_length_mm: The length of the penguin's flipper in mm
             - body_mass_g: The mass of the penguin in grams
             - sex: The sex of the penguin (0 = nan, 1 = female, 2 = male)
    
    Returns:
    prediction (int): The predicted species of the penguin (0 = Adelie, 1 = Chinstrap, 2 = Gentoo)
    nodes (list): A list representing the truth values of the inner nodes
                  where 1 indicates the condition is satisfied and 0 otherwise.
    """
    
    # Initialize the nodes with None
    nodes = 2 * [None]
    
    # Check if the penguin is from Biscoe island
    nodes[0] = X["island"] == 0
    
    # If not from Biscoe, check the flipper length
    nodes[1] = X["flipper_length_mm"] <= 190  # Assuming a threshold of 190 mm for Chinstrap penguins
    
    # Predict the species based on the node conditions
    if nodes[0]:
        prediction = 0  # Adelie
    else:
        if nodes[1]:
            prediction = 1  # Chinstrap
        else:
            prediction = 2  # Gentoo
    
    # Convert boolean nodes to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
