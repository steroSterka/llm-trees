```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score, classification_report

# Assuming this is your dataset with 'party_affiliation' being the target variable
data = {
    'Vote on handicapped infants': ['yes', 'no', 'yes', 'no', 'yes', 'no'],
    'Vote on water project cost sharing': ['yes', 'yes', 'no', 'no', 'yes', 'yes'],
    'Vote on adoption of the budget resolution': ['yes', 'no', 'yes', 'yes', 'no', 'no'],
    # Add all other features here...
    'party_affiliation': ['Democrat', 'Republican', 'Democrat', 'Republican', 'Democrat', 'Republican']
}

df = pd.DataFrame(data)

# Convert categorical variables into numerical variables
for column in df.columns:
    if column != 'party_affiliation':
        df[column] = df[column].map({'yes': 1, 'no': 0})

# Map party affiliation to numeric values for classification
df['party_affiliation'] = df['party_affiliation'].map({'Democrat': 0, 'Republican': 1})

X = df.drop('party_affiliation', axis=1)
y = df['party_affiliation']

# Split dataset into training set and test set
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=1)

# Create Decision Tree classifier with max depth of 2
clf = DecisionTreeClassifier(max_depth=2, random_state=1)

# Train the model
clf.fit(X_train, y_train)

# Predict the response for test dataset
y_pred = clf.predict(X_test)

print("Accuracy:", accuracy_score(y_test, y_pred))
print("Classification Report:\n", classification_report(y_test, y_pred))

# To visualize or understand the decision tree, you can use:
from sklearn.tree import plot_tree

This example uses two features: 'handicapped-infants' and 'el-salvador-adi'. The thresholds used are based on the encoding provided (2 = yes, 1 = no), so a vote in favor is considered as leaning towards one party over the other.

```python
def predict_party_affiliation(X: dict):
    """
    Predicts the party affiliation of an individual based on their voting record.
    
    Parameters:
    X (dict): A dictionary containing the voting records. The keys are the features, and the values are the votes cast (2 = yes, 1 = no).
    
    Returns:
    prediction (int): The predicted party affiliation (0 = Democrat, 1 = Republican).
    nodes (list): A list representing the truth values of the inner nodes.
    """

    # Initialize nodes with None
    nodes = 2 * [None]
    
    # Node 0: Check if the individual voted in favor of handicapped infants
    nodes[0] = X["handicapped-infants"] == 2
    
    # Node 1: Check if the individual voted in favor of aid to El Salvador
    nodes[1] = X["el-salvador-adi"] == 2
    
    # Based on the decision tree, make a prediction
    if nodes[0]:
        # If they voted for handicapped infants, check their stance on El Salvador aid
        if nodes[1]:
            # Voting yes on both suggests a Republican leaning
            prediction = 1
        else:
            # Voting yes on one and no on the other could suggest either party, but let's lean Democrat here
            prediction = 0
    else:
        # Not voting for handicapped infants, check their stance on El Salvador aid
        if nodes[1]:
            # This combination might be less common; for simplicity, let's predict Republican
            prediction = 1
        else:
            # Voting no on both suggests a Democrat leaning
            prediction = 0
    
    # Convert boolean values to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage:
