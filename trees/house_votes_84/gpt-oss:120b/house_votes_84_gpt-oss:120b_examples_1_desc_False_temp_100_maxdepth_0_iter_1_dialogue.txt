```python
def predict(X: dict):
    """
    Predict party affiliation (0 = Democrat, 1 = Republican) from roll‑call votes.

    Parameters
    ----------
    X : dict
        A dictionary that contains **exactly** the keys shown below
        (the values must be the encoded votes: 2 = yes, 1 = no).

        Keys used by the tree
        ---------------------
        'aid-to-nicaraguan-contras'      # 1 = no, 2 = yes
        'mx-missile'                     # 1 = no, 2 = yes
        'education-spending'             # 1 = no, 2 = yes
        'el-salvador-adi'                # 1 = no, 2 = yes
        'handicapped-infants'            # 1 = no, 2 = yes

    Returns
    -------
    prediction : int
        0 → Democrat, 1 → Republican
    nodes : list[int]
        Truth values of the five inner‑node conditions in the order they are
        evaluated. 1 means the condition (the “NO” branch) is satisfied,
        0 means the opposite (“YES”) branch is taken.
    """

    # ------------------------------------------------------------------ #
    # 1️⃣  Root node – aid‑to‑nicaraguan‑contras = NO ?
    # ------------------------------------------------------------------ #
    nodes = [None] * 5                     # placeholder for the 5 inner nodes
    nodes[0] = 1 if X['aid-to-nicaraguan-contras'] == 1 else 0

    # ------------------------------------------------------------------ #
    # 2️⃣  mx‑missile = NO ?
    # ------------------------------------------------------------------ #
    nodes[1] = 1 if X['mx-missile'] == 1 else 0

    # ------------------------------------------------------------------ #
    # 3️⃣  education‑spending = NO ?
    # ------------------------------------------------------------------ #
    nodes[2] = 1 if X['education-spending'] == 1 else 0

    # ------------------------------------------------------------------ #
    # 4️⃣  el‑salvador‑adi = NO ?
    # ------------------------------------------------------------------ #
    nodes[3] = 1 if X['el-salvador-adi'] == 1 else 0

    # ------------------------------------------------------------------ #
    # 5️⃣  handicapped‑infants = NO ?
    # ------------------------------------------------------------------ #
    nodes[4] = 1 if X['handicapped-infants'] == 1 else 0

    # ------------------------------------------------------------------ #
    # Follow the tree logic
    # ------------------------------------------------------------------ #
    if nodes[0]:                     # NO aid to Nicaraguan Contras → Democrat
        prediction = 0
    else:                            # YES aid to Nicaraguan Contras
        if nodes[1]:                 # NO MX missile → Democrat
            prediction = 0
        else:                        # YES MX missile
            if nodes[2]:             # NO education spending → Republican
                prediction = 1
            else:                    # YES education spending
                if nodes[3]:         # NO aid to El Salvador → Democrat
                    prediction = 0
                else:                # YES aid to El Salvador
                    if nodes[4]:     # NO handicapped‑infant funding → Republican
                        prediction = 1
                    else:            # YES handicapped‑infant funding → Democrat
                        prediction = 0

    return prediction, nodes
```
