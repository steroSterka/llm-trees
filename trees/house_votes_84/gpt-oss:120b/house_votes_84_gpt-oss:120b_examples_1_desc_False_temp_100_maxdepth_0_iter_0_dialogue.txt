```python
def predict(X: dict):
    """
    Decision‑tree that predicts party affiliation
        0 → Democrat
        1 → Republican

    Input
    -----
    X : dict
        Keys must match the exact feature names supplied in the prompt
        (e.g. "anti-satellite-test-ban", "mx-missile", …).
        Each vote is encoded as
            2 = yes
            1 = no

    Returns
    -------
    prediction : int
        0 for Democrat, 1 for Republican
    nodes : list[int]
        Truth values of the inner‑node tests (1 = condition true, 0 = false)
        in the order they appear in the tree:
        [anti‑satellite‑test‑ban, mx‑missile,
         aid‑to‑nicaraguan‑contras, duty‑free‑exports,
         education‑spending, superfund‑right‑to‑sue]
    """
    # ------------------------------------------------------------------
    # 1. evaluate every inner‑node condition (store as 0/1 later)
    # ------------------------------------------------------------------
    node0 = X["anti-satellite-test-ban"] == 2          # yes ? 
    node1 = X["mx-missile"] == 2
    node2 = X["aid-to-nicaraguan-contras"] == 2
    node3 = X["duty-free-exports"] == 2
    node4 = X["education-spending"] == 2
    node5 = X["superfund-right-to-sue"] == 2

    # keep the boolean values for the output list
    nodes = [int(node0), int(node1), int(node2),
             int(node3), int(node4), int(node5)]

    # ------------------------------------------------------------------
    # 2. walk the tree
    # ------------------------------------------------------------------
    if node0:                                   # anti‑satellite test ban = YES
        if node1:                               # mx‑missile = YES
            if node2:                           # aid‑to‑Nicaraguan‑Contras = YES
                if node3:                       # duty‑free‑exports = YES
                    if node4:                   # education‑spending = YES
                        # final split on superfund‑right‑to‑sue
                        if node5:               # superfund‑right‑to‑sue = YES
                            prediction = 0      # Democrat
                        else:
                            prediction = 1      # Republican
                    else:
                        prediction = 1          # Republican
                else:
                    prediction = 1              # Republican
            else:
                prediction = 0                  # Democrat
        else:
            prediction = 0                      # Democrat
    else:
        prediction = 0                          # Democrat (anti‑satellite = NO)

    return prediction, nodes
```

The list `nodes` therefore contains a `1` for each test that evaluates to **true** (the legislator voted *yes* on that issue) and `0` otherwise.
