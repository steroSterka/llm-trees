def predict_party_affiliation(X: dict):

        # Node 1: Check aid to Nicaraguan contras vote is 'yes' (2)
        aid_nicaragua_vote = X['aid-to-nicaraguan-contras']
        nodes[1] = aid_nicaragua_vote >= 2
        
        if nodes[1]:
            prediction = 1  # Republican
        else:
            # Node 2: Check education spending vote is 'yes' (2)
            education_spending_vote = X['education-spending']
            nodes[2] = education_spending_vote >= 2
