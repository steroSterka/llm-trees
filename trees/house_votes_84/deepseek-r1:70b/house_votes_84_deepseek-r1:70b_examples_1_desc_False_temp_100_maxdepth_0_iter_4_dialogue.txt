def predict_party_affiliation(X: dict):
    nodes = [None] * 4  # Initialize list to store node evaluations
    
    # Node 0: Check if vote on MX Missile is 'Yes'
    nodes[0] = X["mx-missile"] == 2
    
    # Node 1: Check if vote on Aid to Nicaraguan Contras is 'Yes'
    nodes[1] = X["aid-to-nicaraguan-contras"] == 2
    
    # Node 2: Check if vote on Religious Groups in Schools is 'Yes'
    nodes[2] = X["religious-groups-in-schools"] == 2
    
    # Node 3: Check if vote on Adoption of the Budget Resolution is 'Yes'
    nodes[3] = X["adoption-of-the-budget-resolution"] == 2
    
    prediction = 0  # Default to Democrat (0)
    
    if nodes[0]:
        prediction = 1
    else:
        if nodes[1]:
            prediction = 1
        else:
            if nodes[2]:
                prediction = 1
            else:
                if nodes[3]:
                    prediction = 1
                else:
                    prediction = 0
    
    # Convert boolean values in nodes to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage:
