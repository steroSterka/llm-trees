def predict(X: dict):
    # Evaluate the conditions for each inner node in order
    aid_nicaragua = X['aid-to-nicaraguan-contras'] == 2
    mx_missile = X['mx-missile'] == 2

    if aid_nicaragua:

    else:
        prediction = 0

    return (prediction, nodes)
