def predict(X: dict):

    Predicts party affiliation (Democrat or Republican) based on voting records 
    using a decision tree with depth 2.

                   Keys are the feature names (see problem description).

            - prediction (int): The predicted party affiliation (0 = Democrat, 1 = Republican).
            - nodes (list): A list representing the truth values of the inner nodes 
                             (1 if the condition is satisfied, 0 otherwise).
    """
    
    # Node 0: Aid to Nicaraguan Contras
    node0 = X["aid-to-nicaraguan-contras"] == 2  # 2 = yes, 1 = no

    # Node 1: Education Spending (if aid to contras is no)
    node1 = X["education-spending"] == 2 #2 = yes, 1 = no

    # Node 2: Superfund Right to Sue (if aid to contras is yes)
    node2 = X["superfund-right-to-sue"] == 2 #2 = yes, 1 = no

    nodes = [0, 0, 0]
    nodes[0] = 1 if node0 else 0
    
    if node0:  # Went left (no on aid to contras)
        if node1: # went left again (yes on education spending)
            prediction = 0  # Democrat
        else:
            prediction = 1 # Republican
    else:  # Went right (yes on aid to contras)
        if node2: # went left again (yes on superfund right to sue)
            prediction = 0 # Democrat
        else:
            prediction = 1 #Republican

    nodes[1] = 1 if node1 else 0
    nodes[2] = 1 if node2 else 0
    
    return prediction, nodes
