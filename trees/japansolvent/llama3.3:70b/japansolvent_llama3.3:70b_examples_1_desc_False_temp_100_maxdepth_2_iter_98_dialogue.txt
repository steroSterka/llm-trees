def predict(X: dict):
    """
    Predicts whether a company is solvent or not using a decision tree.

    Args:
    X (dict): A dictionary containing the financial ratios of the company.
        - EBIT/TA (float): Earnings Before Interest and Taxes/Total Assets
        - NI/TC (float): Net Income/Total Capital
        - Sales/TA (float): Sales/Total Assets
        - EBIT/Sales (float): Earnings Before Interest and Taxes/Sales
        - NI/Sales (float): Net Income/Sales
        - WC/TA (float): Working Capital/Total Assets
        - Equity/TL (float): Equity/Total Liabilities
        - Equity/TA (float): Equity/Total Assets

    Returns:
    tuple: A tuple containing the prediction and a list of truth values for the inner nodes.
        - prediction (int): 1 if the company is solvent, 0 otherwise.
        - nodes (list): A list of boolean values representing the conditions at each node.
    """

    # Initialize the nodes with None
    nodes = 2 * [None]

    # Node 0: Check if EBIT/TA is greater than or equal to 0.05
    nodes[0] = X['EBIT/TA'] >= 0.05

    # Node 1: If node 0 is True, check if Equity/TL is greater than 1.0; otherwise, check if NI/TC is greater than 0.10
    if nodes[0]:
        nodes[1] = X['Equity/TL'] > 1.0
    else:
        nodes[1] = X['NI/TC'] > 0.10

    # Make the prediction based on the nodes
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Company is solvent
        else:
            prediction = 0  # Company may be insolvent
    else:
        if nodes[1]:
            prediction = 1  # Company is likely solvent despite low EBIT/TA
        else:
            prediction = 0  # Company is likely insolvent

    # Convert boolean values to integers for the nodes list
    nodes = [int(node) for node in nodes]

    return prediction, nodes
