def predict(X: dict):
    """
    Predicts whether a company is solvent or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the feature values. The keys are 'EBIT/TA', 'NI/TC', 
              'Sales/TA', 'EBIT/Sales', 'NI/Sales', 'WC/TA', 'Equity/TL', 'Equity/TA'.
    
    Returns:
    prediction (int): The predicted class (0 for insolvent, 1 for solvent).
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """
    nodes = 2 * [None]
    # Check if EBIT/TA is less than or equal to 0.10
    nodes[0] = X["EBIT/TA"] <= 0.10
    
    # If EBIT/TA is low, check if Equity/TL is greater than 1.0
    if nodes[0]:
        nodes[1] = X["Equity/TL"] > 1.0
    else:
        # If EBIT/TA is high, check if WC/TA is greater than 0.20
        nodes[1] = X["WC/TA"] > 0.20
    
    # Make the prediction based on the node values
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Solvent if Equity/TL is high and EBIT/TA is low
        else:
            prediction = 0  # Insolvent if Equity/TL is low and EBIT/TA is low
    else:
        if nodes[1]:
            prediction = 1  # Solvent if WC/TA is high and EBIT/TA is high
        else:
            prediction = 0  # Insolvent if WC/TA is low and EBIT/TA is high
    
    return prediction, [int(node) for node in nodes]
