def predict(X: dict):
    """
    Predicts whether a company is solvent or not based on its financial metrics.

    Args:
        X (dict): A dictionary containing the company's financial metrics.
            - EBIT/TA: Earnings Before Interest and Taxes/Total Assets
            - NI/TC: Net Income/Total Capital

    if not nodes[0]:
        nodes[1] = X['EBIT/TA'] <= 0.10

    if not nodes[0] and not nodes[1]:

    # Make a prediction based on the conditions evaluated at each inner node
    if nodes[0]:  # If Equity/TL <= 1.5, predict insolvent (0)
        prediction = 0
    elif nodes[1]:  # If EBIT/TA <= 0.10 and Equity/TL > 1.5, predict insolvent (0)

    return prediction, nodes
