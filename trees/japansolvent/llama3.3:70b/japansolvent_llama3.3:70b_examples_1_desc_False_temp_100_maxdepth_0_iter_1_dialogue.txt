def predict_solvency(X: dict):
    """
    Predicts whether a company is solvent or not based on its financial features.

    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should match the feature names exactly, including case and spacing.

    Returns:
    prediction (int): 1 if the company is predicted to be solvent, 0 otherwise.
    nodes (list): A list of truth values for each inner node in the decision tree.
                  1 indicates the condition was satisfied, 0 indicates it was not.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 3 * [None]

    # Define the conditions for each inner node
    nodes[0] = X['Equity/TL'] > 1.5  # High equity compared to liabilities indicates good solvency
    nodes[1] = X['NI/TC'] > 0.15  # Sufficient net income relative to total capital
    nodes[2] = X['EBIT/TA'] > 0.10  # Healthy operational profitability

    # Make the prediction based on the conditions
    if nodes[0]:
        prediction = 1  # Company is likely solvent with high equity and sufficient income
    else:
        if nodes[1]:
            prediction = 1  # Despite lower equity, sufficient net income suggests solvency
        else:
            if nodes[2]:
                prediction = 1  # Healthy operational profitability indicates potential for solvency
            else:
                prediction = 0  # Insufficient conditions suggest the company is insolvent

    # Convert boolean values in nodes to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
