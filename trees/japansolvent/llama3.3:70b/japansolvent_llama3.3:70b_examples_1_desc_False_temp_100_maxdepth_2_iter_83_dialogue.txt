```python
def predict_financial_solvency(X: dict):
    """
    Predicts whether a company is solvent or not based on its financial features.
    
    Parameters:
    X (dict): Dictionary containing the company's financial features.
             - 'EBIT/TA': Earnings Before Interest and Taxes/Total Assets
             - 'Equity/TL': Equity/Total Liabilities
    
    Returns:
    prediction (int): 1 if the company is predicted to be solvent, 0 otherwise.
    nodes (list): List of boolean values representing the conditions at each inner node.
                  nodes[0] corresponds to the condition 'EBIT/TA <= 0.05',
                  nodes[1] corresponds to the condition 'Equity/TL <= 1.0'.
    """
    
    # Initialize the list to store the truth values of inner nodes
    nodes = 2 * [None]
    
    # First node: Check if EBIT/TA is less than or equal to 5% (a common threshold for profitability)
    nodes[0] = X['EBIT/TA'] <= 0.05
    
    # If the company's profitability is above the threshold, proceed to check leverage
    if not nodes[0]:
        # Second node: Check if Equity/TL is less than or equal to 1 (indicating high debt)
        nodes[1] = X['Equity/TL'] <= 1.0
        
        # Prediction based on the decision tree logic
        if nodes[1]:
            prediction = 0  # Insolvent due to high leverage despite being profitable
        else:
            prediction = 1  # Solvent, given reasonable profitability and low leverage
    else:
        # If the company's profitability is below the threshold, predict insolvency regardless of leverage
        nodes[1] = None  # This node isn't reached in this branch, so mark as None
        prediction = 0  # Insolvent due to low profitability
    
    return prediction, [int(node) if node is not None else None for node in nodes]
```
