def predict(X: dict):
    """
    Predicts whether a company is solvent (1) or insolvent (0) based on its financial features.
    
    Parameters:
    X (dict): Dictionary containing the financial features of the company, including 'EBIT/TA', 'NI/TC', 'Sales/TA', 
              'EBIT/Sales', 'NI/Sales', 'WC/TA', 'Equity/TL', and 'Equity/TA'.
    
    Returns:
    prediction (int): 1 if the company is predicted to be solvent, 0 otherwise.
    nodes (list): List of boolean values indicating whether each inner node condition was satisfied (1) or not (0).
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]
    
    # First node: Check if EBIT/TA is greater than 0.05, which indicates sufficient profitability
    nodes[0] = X['EBIT/TA'] > 0.05
    
    # Second node (only evaluated if the first condition is true): Check if Equity/TL is greater than 1.2, 
    # indicating a healthy equity-to-liabilities ratio
    if nodes[0]:
        nodes[1] = X['Equity/TL'] > 1.2
    else:
        nodes[1] = False
    
    # Make the prediction based on the node conditions
    if nodes[0] and nodes[1]:
        prediction = 1  # Solvent: Both sufficient profitability and a healthy equity-to-liabilities ratio
    elif not nodes[0]: 
        prediction = 0  # Insolvent: Low profitability
    else:
        prediction = 0  # Insolvent: Even with moderate profitability, high leverage (low Equity/TL) indicates insolvency
    
    # Convert boolean values in the nodes list to integers for output consistency
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage:
