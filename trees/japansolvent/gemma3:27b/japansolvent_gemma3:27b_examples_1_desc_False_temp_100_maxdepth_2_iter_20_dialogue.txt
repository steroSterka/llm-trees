def predict(X: dict):
    """
    Predicts company solvency based on financial ratios using a decision tree.

    Args:
        X (dict): A dictionary containing the financial ratios.  Keys must match
                  ['EBIT/TA', 'NI/TC', 'Sales/TA', 'EBIT/Sales', 'NI/Sales',
                   'WC/TA', 'Equity/TL', 'Equity/TA'].

    Returns:
        tuple: A tuple containing:
            - prediction (int): 0 for insolvent, 1 for solvent.
            - nodes (list): A list of boolean values representing the truth
                             values of the inner nodes during the prediction.
    """

    # Initialize the list to store node truth values
    nodes = []

    # Node 1: EBIT/TA <= 0.05
    ebit_ta = X['EBIT/TA']
    node1 = ebit_ta <= 0.05
    nodes.append(node1)

    if node1:
        # Node 2: WC/TA <= 0.10
        wc_ta = X['WC/TA']
        node2 = wc_ta <= 0.10
        nodes.append(node2)

        if node2:
            prediction = 0  # Insolvent
        else:
            prediction = 1  # Solvent
    else:
        # Node 3: Equity/TL <= 0.50
        equity_tl = X['Equity/TL']
        node3 = equity_tl <= 0.50
        nodes.append(node3)

        if node3:
