def predict(X: dict):
    """
    Predicts company solvency based on financial ratios using a decision tree.

    Args:
        X (dict): A dictionary containing the financial ratios for a company.

    """

    # Node 1: Net Income/Total Capital <= 0.05
    nodes[0] = X["NI/TC"] <= 0.05

    if nodes[0]:
        # If NI/TC is low, the company is likely insolvent
        prediction = 0
    else:
        # Node 2: Earnings Before Interest and Taxes/Sales <= 0.10
        nodes[1] = X["EBIT/Sales"] <= 0.10

        if nodes[1]:
