```python
def predict(X: dict):
    """
    Predicts company solvency based on financial features using a decision tree.

    Args:
        X: A dictionary containing the financial features.

    Returns:
        A tuple containing:
            - The prediction (1 = solvent, 0 = insolvent).
            - A list of truth values (1 or 0) representing the conditions
              satisfied at each inner node of the tree.
    """

    nodes = []  # List to store truth values of inner nodes
    prediction = 0  # Default prediction (insolvent)

    # Node 1: Net Income/Sales <= 0.05
    ni_sales = X.get('NI/Sales', 0)  # Handle missing value gracefully
    node1_condition = ni_sales <= 0.05
    nodes.append(int(node1_condition))

    if node1_condition:
        # Node 2: Equity/Total Liabilities <= 2.0
        equity_tl = X.get('Equity/TL', 0)
        node2_condition = equity_tl <= 2.0
        nodes.append(int(node2_condition))

        if node2_condition:
            prediction = 0  # Not Solvent
        else:
            prediction = 1  # Solvent
    else:
        # Node 3: Earnings Before Interest and Taxes/Sales <= 0.15
        ebit_sales = X.get('EBIT/Sales', 0)
        node3_condition = ebit_sales <= 0.15
        nodes.append(int(node3_condition))

        if node3_condition:
            prediction = 1  # Solvent
        else:
            prediction = 0  # Not Solvent

    return prediction, nodes
```
