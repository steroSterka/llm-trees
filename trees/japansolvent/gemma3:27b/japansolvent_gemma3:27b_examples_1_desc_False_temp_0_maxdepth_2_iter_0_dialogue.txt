def predict(X: dict):
    """
    Predicts company solvency based on financial ratios using a decision tree.

    Args:
        X (dict): A dictionary containing the financial ratios for a company.

    Returns:
        tuple: A tuple containing the predicted solvency (0 or 1) and a list 
               representing the truth values of the inner nodes.
    """

    nodes = [None, None]  # Initialize a list to store the truth values of the nodes

    # Node 1: EBIT/TA <= 0.05
    nodes[0] = X["EBIT/TA"] <= 0.05

    # Node 2: NI/TC <= 0.02
    nodes[1] = X["NI/TC"] <= 0.02

    if nodes[0]:  # If EBIT/TA <= 0.05
        if nodes[1]:  # And NI/TC <= 0.02
            prediction = 0  # Not Solvent
        else:
            prediction = 1  # Solvent
    else:  # If EBIT/TA > 0.05
        # Node 3: Equity/TL <= 1.0
        equity_tl_condition = X["Equity/TL"] <= 1.0
        if equity_tl_condition:
            prediction = 0  # Not Solvent
        else:
            prediction = 1  # Solvent
    
    return prediction, nodes
