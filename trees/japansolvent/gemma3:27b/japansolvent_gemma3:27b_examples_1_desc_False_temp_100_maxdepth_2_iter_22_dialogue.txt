    """

    Args:

    Returns:
        tuple: A tuple containing:

    nodes = []  # List to store the truth values of the inner nodes

    # First Split: EBIT/TA <= 0.05
    ebit_ta = X['EBIT/TA']
    node1 = ebit_ta <= 0.05
    nodes.append(node1)

        ni_tc = X['NI/TC']
        node2 = ni_tc <= 0.10

        else:  # Left-Right Branch

    else:  # Right Branch
        # Second Split (Right Branch): Equity/TL <= 1.0
        equity_tl = X['Equity/TL']
