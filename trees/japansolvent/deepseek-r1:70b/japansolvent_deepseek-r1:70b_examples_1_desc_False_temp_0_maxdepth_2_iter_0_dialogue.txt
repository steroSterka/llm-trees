def predict(X: dict):
    # Extract necessary features

    wc_ta = X["WC/TA"]

    # Initialize nodes list with boolean values
    nodes = [
        equity_tl > 0.5,  # First condition

    ]

    # Determine prediction based on the decision tree logic
    if nodes[0]:
        prediction = 1
    else:
        if nodes[1]:
