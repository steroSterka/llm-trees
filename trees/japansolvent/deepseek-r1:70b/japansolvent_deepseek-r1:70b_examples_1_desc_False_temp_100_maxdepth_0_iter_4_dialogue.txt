```python
def predict(X: dict):
    """
    Predicts whether a company is solvent based on financial ratios using a decision tree approach.

    Args:
        X (dict): Dictionary containing financial features of the company. Required keys are:
            - 'EBIT/TA'
            - 'NI/TC'
            - 'Sales/TA'
            - 'EBIT/Sales'
            - 'NI/Sales'
            - 'WC/TA'
            - 'Equity/TL'  # Equity/Total Liabilities
            - 'Equity/TA'

    Returns:
        tuple: (prediction, nodes)
            - prediction (int): 1 if the company is predicted to be solvent, 0 otherwise.
            - nodes (list): List of truth values for inner nodes in the decision tree.
                - 1 if condition satisfied, 0 if not satisfied, None if not applicable.

    """
    # Initialize list to store truth values of inner nodes
    nodes = [None] * 5

    # First split: Equity/Total Liabilities > Threshold A (e.g., 1.0)
    equity_tl_threshold = 1.0
    nodes[0] = X['Equity/TL'] > equity_tl_threshold

    if nodes[0]:
        prediction = 1  # Solvent
    else:
        # Second split: EBIT/Total Assets > Threshold B (e.g., 0.15)
        ebitta_threshold = 0.15
        nodes[1] = X['EBIT/TA'] > ebitta_threshold

        if nodes[1]:
            prediction = 1  # Solvent
        else:
            # Third split: Net Income/Sales > Threshold C (e.g., 0.05)
            ni_sales_threshold = 0.05
            nodes[2] = X['NI/Sales'] > ni_sales_threshold

            if nodes[2]:
                prediction = 1  # Solvent
            else:
                prediction = 0  # Insolvent

    # Additional nodes are set to 0 since they're not used in this tree
    for i in range(3, len(nodes)):
        nodes[i] = 0

    return prediction, [int(node) if node is not None else 0 for node in nodes]
```
