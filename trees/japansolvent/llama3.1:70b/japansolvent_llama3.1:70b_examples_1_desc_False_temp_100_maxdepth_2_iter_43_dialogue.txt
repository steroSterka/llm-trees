def predict(X: dict):
    """
    Predicts whether a company is solvent or not based on its financial features.

    Args:
        X (dict): A dictionary containing the financial features of the company.
            The keys are 'EBIT/TA', 'NI/TC', 'Sales/TA', 'EBIT/Sales', 'NI/Sales',
            'WC/TA', 'Equity/TL', 'Equity/TA'.

    Returns:
        tuple: A tuple containing the predicted class label (1 = solvent, 0 = insolvent)
            and a list of truth values for the inner nodes.
    """

    # Initialize the inner node truth values
    nodes = [None] * 2

    # Split on EBIT/TA feature
    threshold_ebit_ta = 0.12  # Adjust this value based on your dataset
    nodes[0] = X['EBIT/TA'] <= threshold_ebit_ta

    # If the company has low EBIT/TA, split on Equity/TL feature
    threshold_equity_tl = 1.22  # Adjust this value based on your dataset
    if nodes[0]:
        nodes[1] = X['Equity/TL'] <= threshold_equity_tl
    else:
        nodes[1] = False  # Dummy value, won't affect the prediction

    # Make a prediction based on the inner node truth values
    if nodes[0]:
        if nodes[1]:
            prediction = 0  # Insolvent

    else:

    return prediction, nodes
