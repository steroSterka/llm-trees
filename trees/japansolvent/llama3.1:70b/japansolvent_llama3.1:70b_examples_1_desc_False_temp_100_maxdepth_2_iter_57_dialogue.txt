def predict(X: dict):
    """
    Predicts whether a company is solvent or not based on its financial features.

    Args:
        X (dict): A dictionary containing the financial features of the company.
            The keys should match the exact spelling and casing of the feature names.

    Returns:
        tuple: A tuple containing two values:
            prediction (int): 1 if the company is predicted to be solvent, 0 otherwise.
            nodes (list): A list of boolean values representing the truth values of the inner nodes in the decision tree.
    """

    # Define the nodes of the decision tree
    nodes = [None] * 2

    # Node 0: Equity/Total Liabilities <= 1.5
    nodes[0] = X["Equity/TL"] <= 1.5

    # Node 1: Net Income/Total Capital > 0.05 (only considered if node 0 is False)
    nodes[1] = X["NI/TC"] > 0.05

    # Predict the solvency of the company based on the decision tree
    if not nodes[0]:  # If Equity/TL <= 1.5, consider Node 1
        prediction = int(nodes[1])  # Prediction is equal to Node 1 value (True -> 1, False -> 0)
    else:  # Otherwise, company is predicted to be insolvent
        prediction = 0

    return prediction, nodes
