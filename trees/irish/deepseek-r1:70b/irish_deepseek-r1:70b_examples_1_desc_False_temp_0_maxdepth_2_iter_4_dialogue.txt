def predict(X):
    nodes = [None, None]
    
    # Evaluate the first condition: Educational_level >5
    nodes[0] = X['Educational_level'] > 5
    
    # Evaluate the second condition: Type_school is secondary or vocational (>=1)
    nodes[1] = X['Type_school'] >= 1
    
    if nodes[0]:
        prediction = 1
    else:
        if nodes[1]:
            prediction = 1

            prediction = 0

    return prediction, node_truth_values
