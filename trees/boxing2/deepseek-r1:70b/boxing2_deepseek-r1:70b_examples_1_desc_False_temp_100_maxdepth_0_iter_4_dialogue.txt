```python
def predict(X):
    nodes = [None] * 5
    
    # Node 0: Check if the judge is an official one
    nodes[0] = (X["Official"] == 1)
    
    if nodes[0]:
        # If it's an official judge, check specific judges
        nodes[1] = (X["Judge"] in [3, 4, 6])  # HBO-Lederman, Sportsticker, USA Today
        
        if nodes[1]:
            # For these judges, check the round number
            nodes[2] = (X["Round"] <= 6)
            
            if nodes[2]:
                prediction = 0  # Trinidad wins
            else:
                prediction = 1  # de la Hoya wins
        else:
            # Other official judges: G. Hamada, B. Logist, J. Roth, Las Vegas Review-Journal, Los Angeles Times-Springer, van de Wiele
            nodes[3] = (X["Round"] <= 9)
            
            if nodes[3]:
                prediction = 0  # Trinidad wins
            else:
                prediction = 1  # de la Hoya wins
    else:
        # If it's not an official judge, check specific non-official judges
        nodes[4] = (X["Judge"] in [5, 7])  # Los Angeles Times-Kawakami, Associated Press
        
        prediction = 0  # Trinidad wins regardless of the round
    
    return prediction, nodes
```
