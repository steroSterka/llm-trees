   - First, check if 'Official' is 1. If true, predict Trinidad (0) and set nodes[0] = 1.
   - If 'Official' is 0, then check the 'Round':
     - If Round <=6: Predict de la Hoya (1), set nodes[1] =1.

def predict(X: dict) -> tuple[int, list[int]]:
    # Initialize all nodes to 0 (False)
    nodes = [0, 0, 0]
    
    official = X['Official']
    round_num = X['Round']
    
    if official == 1:
        prediction = 0
        nodes[0] = 1
