```python
    else:

        nodes[3] = X["Round"] <= 9
    else:
        nodes[3] = False
    
    # Node 4 is not used in this tree
    
    if nodes[0]:
        if nodes[1]:
            prediction = 0 if nodes[2] else 1
        else:

    else:
        prediction = 0 if nodes[3] else 1

```
