def predict(X: dict):
    """
    Predicts which boxer won the round based on the given features.

    Args:
        X (dict): A dictionary containing the features.
            - 'Judge': The judge's ID (0-10)
            - 'Official': Whether the judge is official (1 = Yes, 0 = No)
            - 'Round': The round number (1-12)

    Returns:
        tuple: A tuple containing the predicted winner and a list of truth values for the inner nodes.

        prediction = 1  # de la Hoya wins late rounds or with non-official judges and high judge IDs
    else:
