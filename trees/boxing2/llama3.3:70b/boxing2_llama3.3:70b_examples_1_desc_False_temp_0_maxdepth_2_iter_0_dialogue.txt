    """
    Predicts which boxer won each round based on the given features.

    Args:
        X (dict): A dictionary containing the features.
            - 'Judge': The judge's ID (0-10).
            - 'Official': Whether the judge is official (1) or not (0).

    if nodes[0]:

    else:
        if nodes[1]:
            prediction = 0  # Trinidad wins in early rounds
        else:

    # Convert boolean node values to integers (1 for True, 0 for False)

    return prediction, nodes
