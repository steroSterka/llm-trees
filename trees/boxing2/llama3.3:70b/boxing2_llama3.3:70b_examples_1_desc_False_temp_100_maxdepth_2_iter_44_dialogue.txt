def predict(X: dict):
    """
    Predicts which boxer won a round based on the judge and round number.

    Args:

            - 'Judge': The judge of the round (0 = G. Hamada, 1 = B. Logist, 2 = J. Roth,

              9 = Los Angeles Times-Springer, 10 = van de Wiele)
            - 'Official': Whether the judge is official (1 = Yes, 0 = No)
            - 'Round': The round number (1 to 12)

    Returns:
        tuple: A tuple containing the prediction and a list of node values.
            - prediction: Which boxer won the round (0 = Trinidad, 1 = de la Hoya)
            - nodes: A list of boolean values representing the inner nodes of the tree

    nodes = 2 * [None]

    # Define the conditions for each node
    prominent_judges = [3, 5, 6, 7]  # HBO-Lederman, Los Angeles Times-Kawakami, USA Today, Associated Press

    nodes[1] = X['Round'] <= 6

    # Make the prediction based on the node values
    if nodes[0]:
        prediction = 0  # Trinidad wins if the judge is prominent

        else:
