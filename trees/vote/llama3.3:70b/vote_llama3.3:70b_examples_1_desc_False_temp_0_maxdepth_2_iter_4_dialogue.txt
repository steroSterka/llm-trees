def predict(X: dict):
    nodes = 2 * [None]

    # Node 1 checks if the politician voted yes on mx missile
    nodes[1] = X["mx missile"] == 2

    if nodes[0]:
        # If they voted yes on el salvador aid, and also voted yes on mx missile, they are likely Republican
        if nodes[1]:
            prediction = 1  # Republican
        else:
            # If they didn't vote yes on mx missile but did on el salvador aid, it's less clear, so default to Democrat
            prediction = 0  # Democrat
    else:
        # If they didn't vote yes on el salvador aid, regardless of their vote on mx missile, they are likely Democrat
        prediction = 0  # Democrat

    return prediction, [int(node) for node in nodes]
