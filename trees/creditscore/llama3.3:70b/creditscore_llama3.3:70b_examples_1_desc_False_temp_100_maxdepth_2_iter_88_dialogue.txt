def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not.
    
    Parameters:
    X (dict): A dictionary containing the applicant's information, including:

        - Income.per.dependent (1.5 to 10)

        - Derogatory.reports (number of derogatory reports)
        
    Returns:
    prediction (int): Whether the application for credit is accepted or not (0 = no, 1 = yes)
    nodes (list): A list representing the truth values of the inner nodes
    """
    
    # Initialize nodes with None
    nodes = 2 * [None]
    
    # Check income per dependent at the root level
    nodes[0] = X['Income.per.dependent'] <= 5
    
    # For branches from the root, consider number of derogatory reports
    nodes[1] = X['Derogatory.reports'] <= 1
    
    # Make prediction based on decision tree logic
    if nodes[0]:

    return prediction, nodes
