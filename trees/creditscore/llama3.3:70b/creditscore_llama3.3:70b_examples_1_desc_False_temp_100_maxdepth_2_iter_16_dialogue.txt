def predict(X: dict):
    """
    Predicts whether an application for credit is accepted or not.
    
    Parameters:
    X (dict): A dictionary containing the features.
        - Age: age (years)
        - Income.per.dependent: income per dependent (1.5 to 10)
        - Monthly.credit.card.exp: monthly credit card expenses ($)
        - Own.home: owning a home (1 = yes, 0 = no)
        - Self.employed: self employed (1 = yes, 0 = no)
        - Derogatory.reports: number of derogatory reports (0, 1, 2, 3, 4, 7)
    
    Returns:
    prediction (int): The predicted class (0 = application rejected, 1 = application accepted).
    nodes (list): A list representing the truth values of the inner nodes.
    """

    # Initialize the nodes list with None
    nodes = 2 * [None]
    
    # Node 1: Check if income per dependent is greater than or equal to 3.5
    nodes[0] = X["Income.per.dependent"] >= 3.5
    
    # If the first condition is true, proceed to check the number of derogatory reports
    if nodes[0]:
        # Node 2: Check if the number of derogatory reports is less than 2
        nodes[1] = X["Derogatory.reports"] < 2
        
        # Prediction based on the second condition
        prediction = int(nodes[1])
    else:
        # If the first condition is false, predict rejection (0)
        prediction = 0
        # Set the second node to None since it was not used
        nodes[1] = False
    
    # Convert boolean values in nodes list to integers (1 for True, 0 for False)
    nodes = [int(node) if node is not None else None for node in nodes]
    
    return prediction, nodes

# Example usage:

This function implements the decision tree based on `Income.per.dependent` and `Derogatory.reports`. The first condition checks if the income per dependent is greater than or equal to 3.5.
