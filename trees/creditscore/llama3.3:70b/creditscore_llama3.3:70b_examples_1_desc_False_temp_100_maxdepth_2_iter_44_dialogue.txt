def predict_credit_application(X: dict):
    """
    Predicts whether a credit application will be accepted based on given features.

    Args:
        X (dict): Dictionary containing feature values.
            - Age (years)
            - Income.per.dependent (income per dependent, 1.5 to 10)
            - Monthly.credit.card.exp (monthly credit card expenses, $)
            - Own.home (owning a home, 1 = yes, 0 = no)
            - Self.employed (self employed, 1 = yes, 0 = no)
            - Derogatory.reports (number of derogatory reports, 0, 1, 2, 3, 4, 7)

    Returns:
        prediction (int): Predicted outcome (0 = application not accepted, 1 = application accepted)
        nodes (list): List representing the truth values of inner nodes
    """
    # Initialize list to store node evaluations
    nodes = 2 * [None]

    # Evaluate first split: Income per dependent <= 3
    nodes[0] = X["Income.per.dependent"] <= 3

    # Evaluate second split based on first split outcome
    if nodes[0]:
        # For lower income, evaluate number of derogatory reports <= 1
        nodes[1] = X["Derogatory.reports"] <= 1
    else:
        # For higher income, evaluate monthly credit card expenses <= $500
        nodes[1] = X["Monthly.credit.card.exp"] <= 500

    # Determine prediction based on node evaluations
    if nodes[0]:
        # Lower income per dependent
        if nodes[1]:
            # Few derogatory reports
            prediction = 1  # Application accepted
        else:
            # Many derogatory reports, high risk
            prediction = 0  # Application not accepted
    else:
        # Higher income per dependent
        if nodes[1]:
            # Manageable debt (low credit card expenses)
            prediction = 1  # Application accepted
        else:
            # High debt obligations
            prediction = 0  # Application not accepted

    # Convert node evaluations to integers for output
    nodes = [int(node) for node in nodes]

    return prediction, nodes
