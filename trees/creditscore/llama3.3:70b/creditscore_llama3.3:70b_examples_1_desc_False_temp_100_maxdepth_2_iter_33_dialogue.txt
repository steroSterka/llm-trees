```python
def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on the input features.
    
    Parameters:
    X (dict): A dictionary containing the input features.
        - 'Age': age in years
        - 'Income.per.dependent': income per dependent (1.5 to 10)
        - 'Monthly.credit.card.exp': monthly credit card expenses ($)
        - 'Own.home': owning a home (1 = yes, 0 = no)
        - 'Self.employed': self employed (1 = yes, 0 = no)
        - 'Derogatory.reports': number of derogatory reports (0, 1, 2, 3, 4, 7)
    
    Returns:
    prediction (int): The predicted outcome (0 = application not accepted, 1 = application accepted).
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """

    # Initialize the nodes list with None values
    nodes = 2 * [None]

    # Node 0: Check if the number of derogatory reports is less than or equal to 1
    nodes[0] = X["Derogatory.reports"] <= 1

    # Node 1: Check if the age is less than or equal to 40 and income per dependent is greater than 3
    nodes[1] = X["Age"] <= 40 and X["Income.per.dependent"] > 3

    # Make predictions based on the node values
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Application accepted if all conditions are met
        else:
            prediction = 0  # Application not accepted if income or age condition is not met
    else:
        prediction = 0  # Application not accepted if derogatory reports exceed 1

    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
```
