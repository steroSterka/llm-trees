def predict_credit(X: dict):
    """
    Predicts whether an application for credit is accepted based on a given set of features.

    Args:
        X (dict): A dictionary containing the features.

            - 'Income.per.dependent': income per dependent

            - 'Own.home': owning a home (1 = yes, 0 = no)
            - 'Self.employed': self employed (1 = yes, 0 = no)

        tuple: A tuple containing the prediction (0 = credit not accepted, 1 = credit accepted) and a list representing the truth values of the inner nodes.
    """

    # Initialize nodes
    nodes = 2 * [None]

    if nodes[0]:
        nodes[1] = X["Income.per.dependent"] >= 5

    # Make prediction based on the nodes
