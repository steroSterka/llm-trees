def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted based on a decision tree.

    Parameters:
    X (dict): Dictionary containing features.
        - 'Age': age in years
        - 'Income.per.dependent': income per dependent (1.5 to 10)
        - 'Monthly.credit.card.exp': monthly credit card expenses ($)
        - 'Own.home': owning a home (1 = yes, 0 = no)
        - 'Self.employed': self employed (1 = yes, 0 = no)
        - 'Derogatory.reports': number of derogatory reports (0, 1, 2, 3, 4, 7)

    Returns:
    tuple: A prediction (0 = not accepted, 1 = accepted) and a list representing the truth values of the inner nodes.
    """

    # Initialize nodes
    nodes = 2 * [None]

    # First node: Number of derogatory reports <= 0
    nodes[0] = X['Derogatory.reports'] <= 0

    # Second node (only considered if first node is False): Income per dependent >= 5
    nodes[1] = X['Income.per.dependent'] >= 5 if not nodes[0] else None

    # Prediction based on the decision tree
    if nodes[0]:
        prediction = 1  # Accepted if no derogatory reports
    elif nodes[1]:  # This condition is only reached if nodes[0] is False, so nodes[1] is guaranteed to be set
        prediction = 0  # Rejected due to past payment issues but has stable income per dependent
    else:

    return prediction, [int(node) for node in nodes]
