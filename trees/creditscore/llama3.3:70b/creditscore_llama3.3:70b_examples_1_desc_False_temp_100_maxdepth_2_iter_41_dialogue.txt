def predict_credit_application(X: dict):

    Predicts whether a credit application will be accepted or not based on the provided features.

        - 'Income.per.dependent': income per dependent (1.5 to 10)

    Returns:

    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]

    nodes[0] = X['Income.per.dependent'] <= 5

    # If income per dependent is low, check monthly credit card expenses
    if nodes[0]:
        # Inner node: Monthly credit card expenses <= $1000
        nodes[1] = X['Monthly.credit.card.exp'] <= 1000
    else:

        nodes[1] = X['Derogatory.reports'] <= 1

    # Make the prediction based on the inner nodes
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Accepted: low income per dependent and manageable credit card expenses

            prediction = 0  # Not accepted: low income per dependent and high credit card expenses
    else:
        if nodes[1]:
