def predict_credit(X: dict):
    """
    This function predicts whether an application for credit is accepted or not.
    
    Parameters:
    X (dict): A dictionary containing the features.
        - Age: age in years
        - Income.per.dependent: income per dependent (1.5 to 10)
        - Monthly.credit.card.exp: monthly credit card expenses ($)
        - Own.home: owning a home (1 = yes, 0 = no)
        - Self.employed: self employed (1 = yes, 0 = no)
        - Derogatory.reports: number of derogatory reports (0, 1, 2, 3, 4, 7)
    
    Returns:
    prediction (int): Whether the application for credit is accepted or not (0 = no, 1 = yes).
    nodes (list): A list representing the truth values of the inner nodes.
    """

    # Initialize the nodes list with None values
    nodes = 2 * [None]

    # First node: Number of derogatory reports <= 0
    nodes[0] = X["Derogatory.reports"] <= 0

    # Second node (if first node is False): Income per dependent >= 5
    if not nodes[0]:
        nodes[1] = X["Income.per.dependent"] >= 5
    else:
        nodes[1] = True  # If no derogatory reports, this node doesn't matter

    # Make the prediction based on the nodes
    if nodes[0]:
        prediction = 1  # Credit accepted if no derogatory reports
    elif nodes[1]:
        prediction = 1  # Credit accepted if income per dependent is sufficient
    else:
        prediction = 0  # Credit not accepted

    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes


# Example usage
