def predict_credit_application(X: dict):
    """
    Predicts whether a credit application is accepted or not based on the input features.

    Args:
    X (dict): A dictionary containing the feature values.
        - Age (years)
        - Income.per.dependent (income per dependent, 1.5 to 10)
        - Monthly.credit.card.exp (monthly credit card expenses, $)
        - Own.home (owning a home, 1 = yes, 0 = no)
        - Self.employed (self employed, 1 = yes, 0 = no)
        - Derogatory.reports (number of derogatory reports, 0, 1, 2, 3, 4, 7)

    Returns:
    tuple: A tuple containing the predicted outcome (0 = not accepted, 1 = accepted) and a list representing the truth values of the inner nodes.
    """

    # Initialize nodes with None
    nodes = 2 * [None]

    # Node 0: Check if income per dependent is less than or equal to 4
    nodes[0] = X["Income.per.dependent"] <= 4

    # Node 1: Check if the number of derogatory reports is less than or equal to 1 (for those with income per dependent > 4)
    nodes[1] = X["Derogatory.reports"] <= 1

    # Make predictions based on node values
    if nodes[0]:
        prediction = 0  # Not accepted (high risk due to low income)
    else:
        if nodes[1]:
            prediction = 1  # Accepted (low risk due to stable financial history)
        else:
            prediction = 0  # Not accepted (high risk due to poor credit history)

    # Convert node values to integers (1 for True, 0 for False) and return the prediction along with node truth values
    nodes = [int(node) for node in nodes]
    return prediction, nodes
