```python
def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on a decision tree.
    
    Parameters:
    X (dict): Dictionary containing features 'Age', 'Income.per.dependent', 'Monthly.credit.card.exp', 
              'Own.home', 'Self.employed', and 'Derogatory.reports'.
    
    Returns:
    tuple: A tuple where the first element is the prediction (0 for not accepted, 1 for accepted) and
           the second element is a list of truth values for the inner nodes.
    """

    # Initialize nodes with None values
    nodes = 2 * [None]
    
    # Node 0 condition: Number of derogatory reports <= 1
    nodes[0] = X['Derogatory.reports'] <= 1
    
    # If node 0 is true, then check the monthly credit card expenses
    if nodes[0]:
        # Node 1 condition: Monthly credit card expenses <= $1000
        nodes[1] = X['Monthly.credit.card.exp'] <= 1000
        
        # Prediction based on node conditions
        if nodes[1]:
            prediction = 1  # Accepted if both conditions are met
        else:
            prediction = 0  # Not accepted if monthly expenses are too high
    else:
        # If more than one derogatory report, directly predict not accepted
        prediction = 0
        nodes[1] = False  # Set node 1 to False since it's not considered in this branch
    
    # Convert boolean values of nodes to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage
```

This function implements a decision tree where the primary factor is the number of derogatory reports (`Derogatory.reports`).
