def predict_credit(X: dict):

    nodes[1] = False

    # Node 2: monthly credit card expenses <= $500 (only if node 0 is True and node 1 is False)
    nodes[2] = False
    if nodes[0] and not nodes[1]:
        nodes[2] = X['Monthly.credit.card.exp'] <= 500
        
    # Node 3: owning a home == yes (only if node 0 is False)
    nodes[3] = False
    if not nodes[0]:
        nodes[3] = X['Own.home'] == 1
        
    # Node 4: income per dependent >= 7 (only if node 0 is False and node 3 is True)
    nodes[4] = False
    if not nodes[0] and nodes[3]:
        nodes[4] = X['Income.per.dependent'] >= 7
        
    # Node 5: default node for prediction
    nodes[5] = False
    
    # Make the prediction based on the inner nodes
    prediction = 0  # Default to not accepted
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Accepted if income per dependent is high and no derogatory reports
        else:
            if nodes[2]:
                prediction = 1  # Accepted if monthly credit card expenses are low and no derogatory reports

    # Convert boolean values in the nodes list to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
