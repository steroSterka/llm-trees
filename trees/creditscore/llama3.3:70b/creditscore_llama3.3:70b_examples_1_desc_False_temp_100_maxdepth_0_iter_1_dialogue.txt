def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on the provided features.

    Parameters:
    X (dict): A dictionary containing the feature values.
        - 'Age': age in years
        - 'Income.per.dependent': income per dependent (range: 1.5 to 10)

        - 'Own.home': owning a home (1 = yes, 0 = no)

    nodes (list): A list representing the truth values of the inner nodes.
    """

    nodes[1] = X['Income.per.dependent'] > 5 and X['Monthly.credit.card.exp'] <= 1000

    nodes[2] = X['Own.home'] == 1 and X['Self.employed'] == 0

    nodes[3] = X['Age'] > 30 and X['Income.per.dependent'] > 3

            else:
                prediction = 0  # Application not accepted
    else:
        if nodes[4] and nodes[5]:

            prediction = 0  # Application not accepted

    return prediction, [int(node) for node in nodes]
