def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not.
    
    Parameters:
    X (dict): A dictionary containing the input features.
        - Age (years)
        - Income.per.dependent (income per dependent, 1.5 to 10)
        - Monthly.credit.card.exp (monthly credit card expenses, $)
        - Own.home (owning a home, 1 = yes, 0 = no)
        - Self.employed (self employed, 1 = yes, 0 = no)
        - Derogatory.reports (number of derogatory reports, 0, 1, 2, 3, 4, 7)
    
    Returns:
    prediction (int): The predicted outcome (0 = application not accepted, 1 = application accepted).
    nodes (list): A list representing the truth values of the inner nodes.
    """
    
    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]
    
    # Define the conditions for the inner nodes
    nodes[0] = X["Income.per.dependent"] <= 5  # Threshold for income per dependent
    nodes[1] = X["Derogatory.reports"] <= 1  # Threshold for number of derogatory reports
    
    # Make predictions based on the node conditions
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Application accepted if low income and few derogatory reports
        else:
            prediction = 0  # Application not accepted if low income and many derogatory reports
    else:
        if X["Own.home"] == 1:  # If high income, check for homeownership
            prediction = 1  # Application accepted if homeowner
        else:
            prediction = 0  # Application not accepted if not homeowner
    
    return prediction, [int(node) for node in nodes]
