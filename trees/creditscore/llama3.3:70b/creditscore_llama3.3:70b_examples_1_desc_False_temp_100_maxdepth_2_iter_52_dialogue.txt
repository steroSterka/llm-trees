def predict_credit(X: dict):
    """
    Predicts whether an application for credit is accepted based on given features.
    
    Parameters:
    X (dict): A dictionary containing feature values. The keys should match the feature names exactly.
             - Age: age (years)
             - Income.per.dependent: income per dependent (1.5 to 10)
             - Monthly.credit.card.exp: monthly credit card expenses ($)
             - Own.home: owning a home (1 = yes, 0 = no)
             - Self.employed: self employed (1 = yes, 0 = no)
             - Derogatory.reports: number of derogatory reports (0, 1, 2, 3, 4, 7)
    
    Returns:
    prediction (int): Whether the application for credit is accepted or not (1 = yes, 0 = no).
    nodes (list): A list representing the truth values of the inner nodes.
                  The first element corresponds to the condition "Derogatory.reports > 0",
                  and the second element corresponds to either "Income.per.dependent >= 5" if the first condition is true,
                  or "Own.home == 1" if the first condition is false.
    """
    
    # Initialize nodes list with None values
    nodes = 2 * [None]
    
    # First node: Check if there are any derogatory reports
    nodes[0] = X["Derogatory.reports"] > 0
    
    # Prediction and second node based on the first node's condition
    if nodes[0]:
        # If there are derogatory reports, check income per dependent
        nodes[1] = X["Income.per.dependent"] >= 5

    # Convert boolean values in nodes to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
