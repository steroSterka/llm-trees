def predict(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on the input features.

    Args:
    X (dict): A dictionary containing the feature values.
        - 'Age': age (years)
        - 'Income.per.dependent': income per dependent (1.5 to 10)
        - 'Monthly.credit.card.exp': monthly credit card expenses ($)
        - 'Own.home': owning a home (1 = yes, 0 = no)
        - 'Self.employed': self employed (1 = yes, 0 = no)
        - 'Derogatory.reports': number of derogatory reports (0, 1, 2, 3, 4, 7)

    Returns:
    tuple: A tuple containing the prediction and a list representing the truth values of the inner nodes.
        - prediction (int): 0 if the application is rejected, 1 if accepted
        - nodes (list): A list where each entry is 1 if the condition of the corresponding inner node is satisfied, and 0 otherwise
    """

    # Initialize the nodes with None
    nodes = 2 * [None]

    # Define the conditions for the inner nodes
    nodes[0] = X["Income.per.dependent"] <= 5  # Consider income per dependent less than or equal to 5 as low
    nodes[1] = X["Derogatory.reports"] > 2  # Consider more than 2 derogatory reports as high risk

    # Make the prediction based on the conditions
    if nodes[0]:  # Low income per dependent
        if nodes[1]:  # High risk due to derogatory reports
            prediction = 0  # Reject the application
        else:
            prediction = 0  # Reject the application due to low income
    else:  # High income per dependent
        if nodes[1]:  # High risk due to derogatory reports
            prediction = 0  # Reject the application
        else:
            prediction = 1  # Accept the application

    # Convert boolean values to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
