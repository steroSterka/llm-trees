def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on given features.
    
    Parameters:
    X (dict): Dictionary containing the features.
        - Age: age in years
        - Income.per.dependent: income per dependent (1.5 to 10)
        - Monthly.credit.card.exp: monthly credit card expenses ($)
        - Own.home: owning a home (1 = yes, 0 = no)
        - Self.employed: self employed (1 = yes, 0 = no)
        - Derogatory.reports: number of derogatory reports (0, 1, 2, 3, 4, 7)
    
    Returns:
    tuple: A tuple containing the prediction (0 for "not accepted", 1 for "accepted") and a list of boolean values representing the inner nodes.
    """

    # First split based on number of derogatory reports

    # If no derogatory reports, further split based on income per dependent and owning a home
    if nodes[0]:
        nodes[1] = (X["Income.per.dependent"] >= 5) or (X["Own.home"] == 1)
    else:
        nodes[1] = False  # If there are derogatory reports, this node's value doesn't matter for prediction
    
    # Prediction based on the decision tree
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Credit accepted if good income or owns a home and no derogatory reports
