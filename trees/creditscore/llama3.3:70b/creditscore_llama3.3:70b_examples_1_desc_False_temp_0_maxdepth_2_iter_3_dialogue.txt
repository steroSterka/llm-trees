def predict_credit_application(X: dict):
    """
    Predicts whether a credit application is accepted based on the given features.

    Args:
        X (dict): A dictionary containing the features of the applicant.
            - Age (years)
            - Income.per.dependent (income per dependent, 1.5 to 10)
            - Monthly.credit.card.exp (monthly credit card expenses, $)
            - Own.home (owning a home, 1 = yes, 0 = no)
            - Self.employed (self employed, 1 = yes, 0 = no)
            - Derogatory.reports (number of derogatory reports, 0, 1, 2, 3, 4, 7)

    Returns:
        prediction (int): The predicted outcome (0 = not accepted, 1 = accepted).
        nodes (list): A list representing the truth values of the inner nodes.
    """

    # Initialize the nodes with None
    nodes = 2 * [None]

    # Node 0: Check if the number of derogatory reports is less than or equal to 0
    nodes[0] = X["Derogatory.reports"] <= 0

    # If there are no derogatory reports, proceed to check income per dependent
    if nodes[0]:
        # Node 1: Check if income per dependent is greater than 3
        nodes[1] = X["Income.per.dependent"] > 3
        # Predict based on the nodes
        prediction = 1 if nodes[1] else 0
    else:
        # If there are derogatory reports, proceed to check age and home ownership
        # Node 1: Check if age is greater than 30 and owning a home is yes (1)
        nodes[1] = X["Age"] > 30 and X["Own.home"] == 1
        # Predict based on the nodes
        prediction = 1 if nodes[1] else 0

    # Convert boolean values to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
