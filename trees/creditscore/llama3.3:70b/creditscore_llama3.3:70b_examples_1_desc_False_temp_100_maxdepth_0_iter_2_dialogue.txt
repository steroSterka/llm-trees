def predict(X: dict):
    """
    Predict whether an application for credit is accepted or not based on a decision tree.
    
    Parameters:
    X (dict): A dictionary containing the features of the application.
              The keys are 'Age', 'Income.per.dependent', 'Monthly.credit.card.exp', 
              'Own.home', 'Self.employed', and 'Derogatory.reports'.
    
    Returns:
    prediction (int): 1 if the application is accepted, 0 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 7 * [None]

    # Node 0: Number of derogatory reports <= 0
    nodes[0] = X['Derogatory.reports'] <= 0

    # If node 0 is true, check if the applicant owns a home
    if nodes[0]:
        # Node 1: Owning a home = yes
        nodes[1] = X['Own.home'] == 1
        if nodes[1]:
            prediction = 1  # Accepted
        else:
            # Node 2: Self employed = no
            nodes[2] = X['Self.employed'] == 0
            if nodes[2]:
                prediction = 1  # Accepted
            else:
                # Node 3: Age >= 30 and income per dependent >= 3
                nodes[3] = X['Age'] >= 30 and X['Income.per.dependent'] >= 3
                if nodes[3]:
                    prediction = 1  # Accepted
                else:
                    prediction = 0  # Rejected
    else:
        # Node 4: Income per dependent >= 5
        nodes[4] = X['Income.per.dependent'] >= 5
        if nodes[4]:
            prediction = 1  # Accepted
        else:
            # Node 5: Monthly credit card expenses <= $500
            nodes[5] = X['Monthly.credit.card.exp'] <= 500
            if nodes[5]:
                prediction = 1  # Accepted
            else:
                prediction = 0  # Rejected
        # Node 6: Derogatory reports > 0 (already true in this branch)
        nodes[6] = True

    return prediction, [int(node) for node in nodes]
