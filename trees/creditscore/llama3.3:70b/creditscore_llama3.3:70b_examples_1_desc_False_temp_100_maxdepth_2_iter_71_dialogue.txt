def predict_credit(X: dict):
    """
    This function predicts whether an application for credit is accepted or not based on a decision tree.
    
    Parameters:
    X (dict): A dictionary containing the features of the applicant. The keys should match the feature names exactly.
    
    Returns:
    prediction (int): 1 if the application is accepted, 0 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes in the decision tree.
    """
    
    # Initialize the nodes list with None
    nodes = 2 * [None]
    
    # Define the conditions for each node
    nodes[0] = X['Derogatory.reports'] <= 1  # Check if the number of derogatory reports is less than or equal to 1
    nodes[1] = X['Income.per.dependent'] > 5  # Check if the income per dependent is greater than 5
    
    # Make the prediction based on the node conditions
    if nodes[0]:
        if nodes[1]:

        else:
            prediction = 0  # Reject the application if income per dependent is low, even with low derogatory reports
    else:
        prediction = 0  # Reject the application if derogatory reports are high
    
    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
