def predict_credit(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on a decision tree.

    Args:
        X (dict): Dictionary containing the input features.
            - 'Age': Age in years
            - 'Income.per.dependent': Income per dependent (1.5 to 10)

            - 'Own.home': Owning a home (1 = yes, 0 = no)
            - 'Self.employed': Self employed (1 = yes, 0 = no)

    Returns:
        prediction (int): Predicted target variable value (0 = application not accepted, 1 = application accepted)
        nodes (list): List of truth values for the inner nodes
    """

    nodes[1] = X['Derogatory.reports'] <= 1  # Condition for node 2: Number of derogatory reports <= 1
