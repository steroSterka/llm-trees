def predict_credit_application(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on the given features.

    Args:
    X (dict): A dictionary containing the feature values.
        - 'Age': age in years
        - 'Income.per.dependent': income per dependent (1.5 to 10)

    """
    nodes = 2 * [None]
    
    # Root node: Number of derogatory reports <= 0
    nodes[0] = X['Derogatory.reports'] == 0
    
    # If no derogatory reports, consider income per dependent > 5
    if nodes[0]:
        nodes[1] = X['Income.per.dependent'] > 5
    else:

        else:

    else:
