def predict(X: dict):
    """
    Predicts whether an application for credit is accepted or not based on a decision tree.
    
    Args:
        X (dict): A dictionary containing the features. Keys are 'Age', 'Income.per.dependent', 
                  'Monthly.credit.card.exp', 'Own.home', 'Self.employed', and 'Derogatory.reports'.
    
    Returns:
        tuple: A tuple containing the predicted class label (0 = rejected, 1 = accepted) and a list representing the truth values of the inner nodes.
    """

    # Define the inner node conditions
    nodes = [None] * 2
    nodes[0] = X["Derogatory.reports"] <= 1
    nodes[1] = X["Income.per.dependent"] > 5

    # Make predictions based on the decision tree
    if nodes[0]:
