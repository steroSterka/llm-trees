    nodes = [None, None]  # Initialize node truth values

    # Make Prediction
    if nodes[0]:

    else:
        if nodes[1]:
            prediction = 1 #Surgery = Yes
        else:
            # Node 3: Abdominal Distension (only checked if pain is not severe and pulse is not increased)
            if X["abdominal_distension"] == 3 or X["abdominal_distension"] == 1: #Severe or moderate
