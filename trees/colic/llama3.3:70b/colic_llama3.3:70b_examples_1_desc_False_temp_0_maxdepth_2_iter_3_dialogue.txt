```python
def predict_horse_colic_surgery(X: dict):
    """
    Predict whether a horse colic lesion is surgical or not based on the degree of pain and abdominal distension.
    
    Parameters:
    X (dict): Dictionary containing features of the horse colic case.
             - 'pain': Degree of Pain (1 = alert, 2 = continuous severe pain, 3 = depressed, 4 = intermittent mild pain, 5 = intermittent severe pain)
             - 'abdominal_distension': Abdominal Distension (1 = moderate, 2 = none, 3 = severe, 4 = slight)
    
    Returns:
    prediction (int): Prediction of whether the horse colic lesion is surgical or not (0 = no surgery, 1 = surgery).
    nodes (list): List of truth values representing the conditions at each inner node.
    """
    
    # Initialize nodes list with None
    nodes = 2 * [None]
    
    # Node 0: Check if the degree of pain indicates severe pain (continuous severe pain or intermittent severe pain)
    nodes[0] = X['pain'] in [2, 5]  # Continuous severe pain or intermittent severe pain
    
    # If severe pain is present, check the abdominal distension
    if nodes[0]:
        # Node 1: Check if abdominal distension is severe
        nodes[1] = X['abdominal_distension'] == 3  # Severe abdominal distension
        
        # Prediction based on nodes
        if nodes[1]:
            prediction = 1  # Surgery likely needed for severe pain and severe abdominal distension
        else:
            prediction = 1  # Surgery might still be considered for severe pain, even without severe abdominal distension
    else:
        # For non-severe pain cases, consider the abdominal distension
        nodes[1] = X['abdominal_distension'] == 3  # Severe abdominal distension
        
        # Prediction based on nodes
        if nodes[1]:
            prediction = 1  # Surgery might be needed for severe abdominal distension, even without severe pain
        else:
            prediction = 0  # Non-surgical approach might be suitable for non-severe pain and non-severe abdominal distension
    
    # Convert boolean values in nodes to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
```
