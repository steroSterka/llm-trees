def predict(X: dict):
    """
    Decision Tree Classifier to predict if a horse colic lesion is surgical or not.
    
    Parameters:
    X (dict): A dictionary containing the feature values.
        - pain: Degree of Pain (1 = alert, 2 = continuous severe pain, 3 = depressed, 
                 4 = intermittent mild pain, 5 = intermittent severe pain)
        - abdominal_distension: Abdominal Distension (1 = moderate, 2 = none, 3 = severe, 4 = slight)
        - nasogastric_reflux: Nasogastric Reflux (0 = more than 1 liter, 1 = less than 1 liter, 
                                               2 = missing, 3 = none)

    Returns:
    prediction (int): Predicted target variable (0 = no surgery, 1 = surgery)
    nodes (list): A list of boolean values representing the truth values of the inner nodes
    """
    
    # Initialize nodes with None
    nodes = 2 * [None]
    
    # Node 1: Check if pain is severe (continuous severe pain or intermittent severe pain)
    nodes[0] = X["pain"] in [2, 5]
    
    # If pain is not severe, check abdominal distension and nasogastric reflux
    if not nodes[0]:
        # Node 2: Check if abdominal distension is severe or nasogastric reflux is more than 1 liter
        nodes[1] = X["abdominal_distension"] == 3 or X["nasogastric_reflux"] == 0
    else:
        # If pain is severe, we don't need to check the second node for this example
        nodes[1] = True
    
    # Make prediction based on nodes
    if nodes[0]:
        prediction = 1  # Surgery required if pain is severe
    elif nodes[1]:
        prediction = 1  # Surgery required if abdominal distension is severe or nasogastric reflux is more than 1 liter
    else:
        prediction = 0  # No surgery required
    
    return prediction, [int(node) for node in nodes]
