def predict(X: dict):
    """
    Predicts whether a company will go bankrupt based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities
    
    Returns:
    tuple: A tuple containing the prediction (0 = no bankruptcy, 1 = bankruptcy) and a list representing the truth values of the inner nodes.
    """

    # Initialize nodes with None
    nodes = 2 * [None]
    
    # Node 0 condition: Working Capital/Total Assets <= 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # If the company has poor liquidity, check its retained earnings
    if nodes[0]:
        # Node 1 condition: Retained Earnings/Total Assets <= 0.00
        nodes[1] = X['RE/TA'] <= 0.00
        
        # Prediction based on conditions
        if nodes[1]:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt
    else:
        # If the company has decent liquidity, check its profitability (EBIT/Total Assets)
        nodes[1] = X['EBIT/TA'] <= 0.05
        
        # Prediction based on conditions
        if nodes[1]:
            prediction = 1  # Bankrupt due to poor profitability
        else:
            prediction = 0  # Not Bankrupt
    
    # Convert boolean node values to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage
