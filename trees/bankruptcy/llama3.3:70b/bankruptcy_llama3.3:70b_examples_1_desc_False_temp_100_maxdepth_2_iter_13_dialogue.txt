```python
def predict(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should match the feature names exactly, including case and spacing.
              
    Returns:
    prediction (int): 0 if the company is predicted not to go bankrupt, 1 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
                  1 if the condition of the corresponding inner node is satisfied, 0 otherwise.
    """
    
    # Initialize the nodes list with None
    nodes = 2 * [None]
    
    # Check if Working Capital/Total Assets is less than or equal to 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # If WC/TA is low, check if EBIT/TA is less than or equal to 0.05
    if nodes[0]:
        nodes[1] = X['EBIT/TA'] <= 0.05
    else:
        # If WC/TA is not low, the second node's condition doesn't matter for prediction
        nodes[1] = X['EBIT/TA'] <= 0.10
    
    # Make the prediction based on the nodes
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt
    else:
        prediction = 0  # Not Bankrupt
    
    # Convert boolean values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage:
```
