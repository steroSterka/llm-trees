def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
             The keys should include 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', and 'BVE/BVL'.
             
    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 if it went bankrupt.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """
    
    # Initialize the nodes with None
    nodes = 2 * [None]
    
    # Define the conditions for the inner nodes
    nodes[0] = X['WC/TA'] <= 0.20  # Check if working capital to total assets ratio is low
    nodes[1] = X['RE/TA'] <= 0.10  # Check if retained earnings to total assets ratio is low
    
    # Make the prediction based on the node conditions
    if not nodes[0]:  # If working capital to total assets ratio is high
        if nodes[1]:  # But retained earnings to total assets ratio is low
            prediction = 1  # Predict bankruptcy
        else:  # Both ratios are high
            prediction = 0  # Predict no bankruptcy
    else:  # Working capital to total assets ratio is low
        prediction = 1  # Predict bankruptcy
    
    # Convert boolean nodes to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
