def predict_bankruptcy(X: dict):
    """
    Predicts whether a company went bankrupt or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
             The keys should be 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'.
    
    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 if it went bankrupt.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """
    
    # Initialize the nodes list with None
    nodes = 2 * [None]
    
    # Node 0: Check if Working Capital/Total Assets is less than or equal to 0.1
    # This threshold is chosen based on common practices in financial analysis,
    # where a low working capital ratio may indicate liquidity issues.
    nodes[0] = X['WC/TA'] <= 0.1
    
    # Node 1: Check if Earnings Before Interest and Tax/Total Assets is less than or equal to 0.05
    # This threshold is chosen based on common practices in financial analysis,
    # where a low EBIT/TA ratio may indicate profitability issues.
    nodes[1] = X['EBIT/TA'] <= 0.05
    
    # Make predictions based on the node values
    if nodes[0]:
        prediction = 1  # If WC/TA is low, predict bankruptcy
    else:
        if nodes[1]:
            prediction = 1  # If EBIT/TA is low and WC/TA is not low, still predict bankruptcy
        else:
            prediction = 0  # If neither condition is met, predict no bankruptcy
    
    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
