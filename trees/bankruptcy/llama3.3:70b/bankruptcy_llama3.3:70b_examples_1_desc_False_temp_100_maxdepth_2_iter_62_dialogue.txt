def predict_bankruptcy(X: dict):
    """
    Predicts whether a company went bankrupt based on its financial features.
    
    Parameters:
    X (dict): Dictionary containing financial features of the company.
             Keys: 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'
    
    Returns:

    """

    nodes = 2 * [None]

    nodes[0] = X['WC/TA'] <= 0.20

        nodes[1] = False  # This node's value doesn't affect the prediction in this case
        prediction = 0
    
    return prediction, [int(node) for node in nodes]
