def predict(X: dict):
    """
    Predict whether a company went bankrupt based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities
    
    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 if it did.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]

    # First split: Working Capital/Total Assets <= 0.20
    nodes[0] = X['WC/TA'] <= 0.20

    # Second split depends on the first split
    if nodes[0]:
        # If the company has liquidity issues, check its retained earnings
        nodes[1] = X['RE/TA'] <= -0.10
    else:
        # If the company does not have liquidity issues, check its EBIT
        nodes[1] = X['EBIT/TA'] <= 0.05

    # Make a prediction based on the splits
    if nodes[0]:
        if nodes[1]:
            # Company has both liquidity and profitability issues
            prediction = 1
        else:
            # Company has liquidity issues but not severe profitability issues
            prediction = 0
    else:
        if nodes[1]:
            # Company does not have liquidity issues but has profitability issues
            prediction = 1
        else:
            # Company does not have liquidity or profitability issues
            prediction = 0

    # Convert boolean values to integers for the nodes list
    nodes = [int(node) for node in nodes]

    return prediction, nodes
