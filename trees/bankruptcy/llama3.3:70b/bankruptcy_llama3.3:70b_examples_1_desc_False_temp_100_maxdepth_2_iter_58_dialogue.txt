def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should include 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', and 'BVE/BVL'.
    
    Returns:
    prediction (int): 0 if the company is predicted not to go bankrupt, 1 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """

    # Define the thresholds for the decision tree
    wc_ta_threshold = 0.20
    re_ta_threshold = 0.10

    # Initialize the nodes with None
    nodes = 2 * [None]

    # Calculate the node values based on the given features and thresholds
    nodes[0] = X['WC/TA'] <= wc_ta_threshold
    nodes[1] = X['RE/TA'] <= re_ta_threshold

    # Make predictions based on the node values
    if not nodes[0]:  # If working capital to total assets is high
        if nodes[1]:  # And retained earnings to total assets is low
            prediction = 1  # Predict bankruptcy
        else:  # Retained earnings to total assets is high
            prediction = 0  # Predict no bankruptcy
    else:  # Working capital to total assets is low
        prediction = 1  # Predict bankruptcy due to liquidity issues

    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes


# Example usage:
features = {
    'WC/TA': 0.15,
    'RE/TA': 0.05,

This function takes a dictionary `X` containing the financial features of a company and returns a prediction (0 for no bankruptcy, 1 for bankruptcy) along with a list of node values representing the truth values of the inner nodes in the decision tree.
