def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt based on its financial metrics.
    
    Parameters:
    X (dict): Dictionary containing the financial metrics of the company.
             The keys should include 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'.
             
    Returns:
    prediction (int): 0 if the company is predicted not to go bankrupt, 1 otherwise.
    nodes (list): List of boolean values representing the conditions at each inner node.

    """
    
    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]
    
    # Root Node: Check if Book Value of Equity/Book Value of Liabilities is less than or equal to 0.5
    nodes[0] = X['BVE/BVL'] <= 0.5
    
    # If the company's BVE/BVL ratio is bad, check its liquidity (WC/TA)
    if nodes[0]:
        nodes[1] = X['WC/TA'] < 0.1
    else:
        # For companies with a better BVE/BVL ratio, also consider their WC/TA but with a different threshold
        nodes[1] = X['WC/TA'] < 0.2
    
    # Make the prediction based on the conditions
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # High risk of bankruptcy due to poor solvency and liquidity
        else:
            prediction = 1  # At risk but slightly better liquidity
    else:
        if nodes[1]:
            prediction = 0  # Survived with challenges (better solvency, manageable liquidity issues)
        else:
            prediction = 0  # Survived (good solvency and liquidity)
    
    # Convert boolean values in nodes to integers for consistency with the problem statement
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
