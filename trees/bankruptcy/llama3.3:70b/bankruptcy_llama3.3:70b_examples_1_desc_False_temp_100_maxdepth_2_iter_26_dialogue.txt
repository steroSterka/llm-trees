def predict(X: dict):

        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities

    Returns:

    """

    # Initialize the nodes with None
    nodes = 2 * [None]

    nodes[0] = X['WC/TA'] <= 0.20

    # Otherwise, this node will not be considered in the prediction

    # Make the prediction based on the nodes
    if nodes[0]:

            prediction = 1  # Bankrupt if Retained Earnings/Total Assets is low despite sufficient Working Capital
