def predict(X: dict):
    """
    This function predicts whether a company went bankrupt or not based on its financial features.

    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should be 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', and 'BVE/BVL'.

    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 if it did.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
                  The first element corresponds to the condition BVE/BVL <= 1,
                  and the second element corresponds to the condition WC/TA <= 0.2 or EBIT/TA <= 0.05.
    """
    nodes = 2 * [None]
    
    # Check if Book Value of Equity/Book Value of Liabilities is less than or equal to 1
    nodes[0] = X['BVE/BVL'] <= 1
    
    # If the company has high leverage, check its liquidity (WC/TA) and profitability (EBIT/TA)
    if nodes[0]:
        nodes[1] = X['WC/TA'] <= 0.2
    else:
        nodes[1] = X['EBIT/TA'] <= 0.05

    # Make a prediction based on the conditions
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # High risk of bankruptcy due to low liquidity or poor profitability and high leverage
        else:
            prediction = 0  # Monitor closely, as the company has high leverage but sufficient liquidity
    else:
        if nodes[1]:
            prediction = 1  # Potential risk due to poor operational efficiency despite lower leverage
        else:
            prediction = 0  # Not bankrupt, as the company has lower leverage and is operationally efficient

    return prediction, [int(node) for node in nodes]
