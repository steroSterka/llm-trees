def predict(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial health indicators.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should match the exact spelling and case of the feature names.
              Required features: 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'
    
    Returns:
    prediction (int): A binary classification indicating whether the company will go bankrupt (1) or not (0).

    nodes = 2 * [None]

    return prediction, nodes
