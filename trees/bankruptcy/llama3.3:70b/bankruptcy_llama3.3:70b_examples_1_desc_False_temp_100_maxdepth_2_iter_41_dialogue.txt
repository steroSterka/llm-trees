def predict(X: dict):
    """
    Predicts whether a company will go bankrupt based on its financial ratios.

    Args:
    X (dict): A dictionary containing the financial ratios of the company.
              The keys should be 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'.

    Returns:
    prediction (int): 0 if the company will not go bankrupt, 1 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
                  The first element represents the condition 'WC/TA <= 0.20' and the second element represents 'BVE/BVL <= 1.00'.
    """
    nodes = 2 * [None]
    # Check if working capital to total assets ratio is less than or equal to 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # If the company has low liquidity, check its solvency
    if nodes[0]:
        # Check if book value of equity to book value of liabilities ratio is less than or equal to 1.00
        nodes[1] = X['BVE/BVL'] <= 1.00
        # If the company has poor solvency, predict bankruptcy
        prediction = 1
    else:
        # If the company has better liquidity, assume it will not go bankrupt
        nodes[1] = False  # This node is not actually used in this case
        prediction = 0

    # Convert boolean values to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
