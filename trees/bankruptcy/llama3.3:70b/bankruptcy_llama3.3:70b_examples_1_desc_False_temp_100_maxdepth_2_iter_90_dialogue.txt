def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should include 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', and 'BVE/BVL'.
              
    Returns:
    prediction (int): 0 if the company is predicted not to go bankrupt, 1 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
                  The first element represents the condition 'BVE/BVL <= 1.0', and the second element represents the condition 'WC/TA <= 0.2'.
    """
    
    # Initialize the nodes list with None values
    nodes = 2 * [None]
    
    # Evaluate the conditions for each node
    nodes[0] = X['BVE/BVL'] <= 1.0  # Check if Book Value of Equity/Book Value of Liabilities is less than or equal to 1.0
    if nodes[0]:
        nodes[1] = X['WC/TA'] <= 0.2  # If the first condition is true, check if Working Capital/Total Assets is less than or equal to 0.2
    else:
        nodes[1] = False  # If the first condition is false, set the second node to False
    
    # Make a prediction based on the node conditions
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # If both conditions are true, predict bankruptcy
        else:
            prediction = 0  # If only the first condition is true, predict no bankruptcy
    else:
        prediction = 0  # If neither condition is true, predict no bankruptcy
    
    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
