This decision tree is a simplified model based on general financial principles:
- Low working capital might indicate difficulties in meeting short-term obligations.
- Negative or low retained earnings could signal persistent losses or insufficient reinvestment, making it harder for a company to survive.

def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on its financial features.

    Args:
        X (dict): Dictionary containing the financial features of the company.
            - 'WC/TA': Working Capital/Total Assets
            - 'RE/TA': Retained Earnings/Total Assets
            - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
            - 'S/TA': Sales/Total Assets
            - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities

    Returns:
        tuple: A tuple containing the prediction (0 = no, 1 = yes) and a list representing the truth values of the inner nodes.
    """

    # Initialize nodes as None
    nodes = 2 * [None]

    # Node 0 condition: Working Capital/Total Assets <= 0.20
    nodes[0] = X['WC/TA'] <= 0.20

    # Node 1 condition: Retained Earnings/Total Assets <= 0.10 (only evaluated if node 0 is False)
    nodes[1] = X['RE/TA'] <= 0.10 if not nodes[0] else None

    # Make prediction based on the decision tree
    if nodes[0]:
        prediction = 1  # Company went bankrupt
    elif nodes[1]:
        prediction = 1  # Company went bankrupt
    else:
        prediction = 0  # Company did not go bankrupt

    # Replace None values in nodes with 0 (to represent the condition was not evaluated)
    nodes = [int(node) if node is not None else 0 for node in nodes]

    return prediction, nodes


# Example usage
example_company = {

    'RE/TA': -0.05,
    'EBIT/TA': 0.10,

}

prediction, nodes = predict(example_company)
print(f"Prediction: Company went bankrupt = {prediction}")
