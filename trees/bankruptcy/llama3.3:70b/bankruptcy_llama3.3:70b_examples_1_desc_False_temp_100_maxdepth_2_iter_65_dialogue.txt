def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial features.

    Parameters:
    X (dict): A dictionary containing the company's financial features.
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities

    Returns:
    tuple: A tuple containing the prediction (0 for no bankruptcy, 1 for bankruptcy)
           and a list representing the truth values of the inner nodes.
    """
    # Initialize nodes with None
    nodes = 2 * [None]
    
    # Node conditions based on financial health indicators
    # Here, we're using hypothetical threshold values for demonstration purposes
    # In practice, these would be determined through data analysis
    nodes[0] = X['WC/TA'] <= 0.20  # Low working capital to total assets ratio
    nodes[1] = X['EBIT/TA'] <= 0.05  # Poor profitability
    
    prediction = 0  # Initialize prediction as no bankruptcy (0)
    
    if nodes[0]:  # If working capital condition is met
        if nodes[1]:  # And if EBIT condition is also met, predict bankruptcy
            prediction = 1
    else:  # If not meeting the first condition but poor profitability, still predict bankruptcy
        if nodes[1]:
            prediction = 1
    
    # Convert boolean values of nodes to integers (True -> 1, False -> 0)
    node_values = [int(node) for node in nodes]
    
    return prediction, node_values
