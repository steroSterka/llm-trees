def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities
    
    Returns:
    prediction (int): The predicted target variable (0 = no, 1 = yes)
    nodes (list): A list representing the truth values of the inner nodes
    """

    # Initialize nodes with None
    nodes = 2 * [None]
    
    # Node 0: Check if Working Capital/Total Assets is less than or equal to 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # If Node 0 is True, check Book Value of Equity/Book Value of Liabilities
    # Otherwise, check Earnings Before Interest and Tax/Total Assets
    if nodes[0]:
        nodes[1] = X['BVE/BVL'] <= 1.00
    else:
        nodes[1] = X['EBIT/TA'] <= 0.10
    
    # Make prediction based on the nodes
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt
    else:
        if nodes[1]:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt
    
    # Convert boolean nodes to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
