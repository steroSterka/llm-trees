def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should match the feature names exactly, including case and spaces.

    Returns:

    nodes (list): A list of boolean values representing the truth values of the inner nodes.
                  The first element corresponds to the condition 'WC/TA <= 0.20', and the second element corresponds to 'EBIT/TA <= 0.05'.
    """
    
    # Initialize the nodes list with None values
    nodes = 2 * [None]

    nodes[0] = X['WC/TA'] <= 0.20  # Condition for the first inner node
    nodes[1] = X['EBIT/TA'] <= 0.05  # Condition for the second inner node
    
    # Make a prediction based on the node conditions

    else:  # If Working Capital/Total Assets is not low
        if nodes[1]:  # If EBIT/TA is low

    return prediction, nodes
