def predict(X: dict):
    """
    This function predicts whether a company went bankrupt or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company. The keys should be:
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities
    
    Returns:
    prediction (int): A binary value indicating whether the company went bankrupt (1) or not (0).
    nodes (list): A list of boolean values representing the truth values of the inner nodes in the decision tree.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]

    # Define the conditions for each node
    nodes[0] = X['EBIT/TA'] <= 0.05  # Condition for the first inner node
    nodes[1] = X['WC/TA'] <= 0.20   # Condition for the second inner node

    # Make the prediction based on the decision tree
    if not nodes[0]:  # If EBIT/TA > 0.05
        if not nodes[1]:  # If WC/TA > 0.20
            prediction = 0  # Company is likely to be solvent (not bankrupt)
        else:  # If WC/TA <= 0.20
            prediction = 1  # Company may face liquidity issues, so it's likely to be bankrupt
    else:  # If EBIT/TA <= 0.05
        prediction = 1  # Company has poor operational efficiency, so it's likely to be bankrupt

    # Convert the boolean values in the nodes list to integers (True -> 1, False -> 0)

    return prediction, nodes
