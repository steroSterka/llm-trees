def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on its financial features.

    Args:
    X (dict): A dictionary containing the company's financial features, including:
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities

    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 otherwise.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """

    # Initialize the nodes with None
    nodes = 2 * [None]

    # Node 0: EBIT/TA <= 0.05 (low profitability)
    nodes[0] = X['EBIT/TA'] <= 0.05

    # If the company has low profitability, check its liquidity (WC/TA)
    if nodes[0]:

        # If the company has poor liquidity, predict bankruptcy
        prediction = 1 if nodes[1] else 0
    else:
        # If the company has decent profitability, predict no bankruptcy
        prediction = 0
        # Set Node 1 to False (not applicable)
        nodes[1] = False

    # Convert boolean values to integers (0 or 1) for nodes
    nodes = [int(node) if node is not None else 0 for node in nodes]

    return prediction, nodes
