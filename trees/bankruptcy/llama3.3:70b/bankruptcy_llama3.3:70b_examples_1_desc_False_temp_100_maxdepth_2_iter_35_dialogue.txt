def predict(X: dict):
    """
    Decision Tree Classifier to predict company bankruptcy.

    Parameters:
    X (dict): Dictionary containing financial features.
        - WC/TA: Working Capital/Total Assets
        - RE/TA: Retained Earnings/Total Assets
        - EBIT/TA: Earnings Before Interest and Tax/Total Assets
        - S/TA: Sales/Total Assets
        - BVE/BVL: Book Value of Equity/Book Value of Liabilities

    Returns:
    prediction (int): Predicted target variable (0 = no bankruptcy, 1 = bankruptcy)
    nodes (list): List of truth values for inner nodes in the decision tree
    """
    # Initialize list to store node conditions
    nodes = 2 * [None]
    
    # Node 1 condition: Working Capital/Total Assets <= 0.20
    nodes[0] = X["WC/TA"] <= 0.20
    
    # If company has low liquidity, evaluate profitability
    if nodes[0]:
        # Node 2 condition: EBIT/TA <= 0.05 (poor profitability)
        nodes[1] = X["EBIT/TA"] <= 0.05
    else:
        # For companies with better liquidity, consider them less likely to go bankrupt regardless of this node's condition
        nodes[1] = False
    
    # Make prediction based on node conditions
    if nodes[0]:
        if nodes[1]:
            prediction = 1  # Bankruptcy likely due to poor liquidity and profitability
        else:
            prediction = 1  # Still at risk despite better profitability, given low liquidity
    else:
        if nodes[1]:  # This condition won't be met due to the previous logic but included for completeness
            prediction = 0  # Not at risk of bankruptcy with good liquidity and some profitability
        else:
            prediction = 0  # Company has better liquidity, reducing bankruptcy risk
    
    return prediction, [int(node) for node in nodes]
