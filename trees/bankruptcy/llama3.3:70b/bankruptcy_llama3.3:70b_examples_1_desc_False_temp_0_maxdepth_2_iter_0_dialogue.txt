def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company, including:
        - 'WC/TA': Working Capital/Total Assets
        - 'RE/TA': Retained Earnings/Total Assets
        - 'EBIT/TA': Earnings Before Interest and Tax/Total Assets
        - 'S/TA': Sales/Total Assets
        - 'BVE/BVL': Book Value of Equity/Book Value of Liabilities
    
    Returns:
    prediction (int): A binary value indicating whether the company is predicted to go bankrupt (1) or not (0).
    nodes (list): A list of boolean values representing the truth values of the inner nodes in the decision tree.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]
    
    # Node 0: Check if Working Capital/Total Assets is less than or equal to 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # Node 1: Check if EBIT/Total Assets is less than or equal to 0.10 (only considered if Node 0 is False)
    nodes[1] = X['EBIT/TA'] <= 0.10
    
    # Make the prediction based on the decision tree
    if nodes[0]:
        prediction = 1  # Predict bankruptcy if Working Capital/Total Assets is low
    else:
        if nodes[1]:
            prediction = 1  # Predict bankruptcy if EBIT/Total Assets is low (given Working Capital/Total Assets is not low)
        else:
            prediction = 0  # Predict no bankruptcy if both conditions are not met
    
    # Convert boolean values in nodes to integers (1 for True, 0 for False) before returning
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
