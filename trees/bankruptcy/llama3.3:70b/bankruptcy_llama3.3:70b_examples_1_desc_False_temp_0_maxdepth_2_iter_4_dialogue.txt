def predict_bankruptcy(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial features.

    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should match the feature names exactly, including case and spaces.
              Required keys: 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'

    Returns:
    prediction (int): 0 if the company is predicted not to go bankrupt, 1 otherwise
    nodes (list): A list of boolean values representing the truth values of the inner nodes
                  in the decision tree. 1 represents True and 0 represents False.
    """

    # Initialize the nodes list with None values
    nodes = 2 * [None]

    # Determine the threshold values for each feature based on common financial analysis practices
    wc_ta_threshold = 0.2  # Companies with WC/TA below this threshold are considered to be at high risk of bankruptcy
    ebit_ta_threshold = 0.1  # Companies with EBIT/TA below this threshold are considered to be unprofitable

    # Calculate the node values based on the feature thresholds
    nodes[0] = X['WC/TA'] <= wc_ta_threshold
    nodes[1] = X['EBIT/TA'] <= ebit_ta_threshold

    # Make the prediction based on the node values
    if nodes[0]:
        prediction = 1  # If WC/TA is below the threshold, predict bankruptcy
    else:
        if nodes[1]:
            prediction = 1  # If EBIT/TA is below the threshold and WC/TA is above, still predict bankruptcy due to unprofitability
        else:
            prediction = 0  # If both conditions are met, predict no bankruptcy

    # Convert boolean node values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
