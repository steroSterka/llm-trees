def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on its financial ratios.

    Parameters:
    X (dict): Dictionary containing financial ratios of a company.
             Keys: 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'

    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 otherwise
    nodes (list): List representing truth values of inner nodes in the decision tree
                  1 if condition is satisfied, 0 otherwise
    """
    # Initialize list to store node conditions
    nodes = 2 * [None]
    
    # Root Node: Working Capital/Total Assets <= 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # Depth 1 Node: Retained Earnings/Total Assets <= 0.10 (if WC/TA condition is met)
    # Otherwise, consider Sales/Total Assets for non-bankrupt classification refinement
    if nodes[0]:
        nodes[1] = X['RE/TA'] <= 0.10
    else:
        nodes[1] = X['S/TA'] >= 1.5
    
    # Prediction logic based on node conditions
    if nodes[0]:  # Poor liquidity
        if nodes[1]:  # Low profitability
            prediction = 1  # High risk of bankruptcy
        else:  # Moderate profitability despite poor liquidity
            prediction = 0  # Lower risk of bankruptcy
    else:  # Better liquidity
        prediction = 0  # Lower risk of bankruptcy
    
    return prediction, [int(node) for node in nodes]
