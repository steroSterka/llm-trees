def predict(X: dict):
    nodes = 2 * [None]
    nodes[0] = X['WC/TA'] <= 0.20

    if nodes[0]:
        nodes[1] = X['RE/TA'] <= 0.10
    else:
        nodes[1] = X['S/TA'] >= 1.5

    if nodes[0]:
        if nodes[1]:
            prediction = 1
        else:
            prediction = 0
    else:
        prediction = 0

    return prediction, [int(node) for node in nodes]
