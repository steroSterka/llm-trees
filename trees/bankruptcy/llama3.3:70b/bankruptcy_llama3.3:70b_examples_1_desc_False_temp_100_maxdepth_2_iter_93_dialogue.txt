def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on its financial features.

    Args:
    X (dict): A dictionary containing the financial features of the company.
              The keys should be 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', and 'BVE/BVL'.

    Returns:
    tuple: A tuple containing the prediction (0 or 1) and a list of truth values for the inner nodes.
    """

    # Initialize the list to store the truth values of the inner nodes
    nodes = 2 * [None]

    # The first node checks if Working Capital/Total Assets is less than or equal to 0.20
    nodes[0] = X['WC/TA'] <= 0.20

    # If the company passes the first check, the second node checks if Retained Earnings/Total Assets is greater than 0.10
    # Otherwise, it checks if Retained Earnings/Total Assets is less than or equal to 0.10
    nodes[1] = X['RE/TA'] <= 0.10 if nodes[0] else X['RE/TA'] > 0.10

    # Make the prediction based on the nodes
    if nodes[0]:
        prediction = 1 if nodes[1] else 0
    else:
        prediction = 0

    # Convert the boolean values in the nodes list to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
