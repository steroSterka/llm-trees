def predict(X: dict):
    """
    Predicts whether a company went bankrupt or not based on its financial features.

    Parameters:
    X (dict): Dictionary containing the financial features of the company.
              The keys are 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', and 'BVE/BVL'.

    Returns:
    prediction (int): 0 if the company did not go bankrupt, 1 if it went bankrupt.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
                  The first element corresponds to the condition Working Capital/Total Assets <= 0.2,
                  and the second element corresponds to the condition Retained Earnings/Total Assets <= -0.5
                  or EBIT/Total Assets > 0.1, depending on the path taken in the decision tree.
    """
    nodes = 2 * [None]
    # Check if Working Capital/Total Assets is less than or equal to 0.2
    nodes[0] = X['WC/TA'] <= 0.2
    
    if nodes[0]:
        # If true, check if Retained Earnings/Total Assets is less than or equal to -0.5
        nodes[1] = X['RE/TA'] <= -0.5
        prediction = 1  # Likely to go bankrupt
    else:
        # If false, check if EBIT/Total Assets is greater than 0.1
        nodes[1] = X['EBIT/TA'] > 0.1
        prediction = 0  # Less likely to go bankrupt

    return prediction, [int(node) for node in nodes]

# Example usage:
X = {
    'WC/TA': 0.15,
    'RE/TA': -0.6,
    'EBIT/TA': 0.05,
