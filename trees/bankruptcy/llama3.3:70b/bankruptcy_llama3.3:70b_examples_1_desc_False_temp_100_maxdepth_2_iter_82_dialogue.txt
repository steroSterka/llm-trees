def predict(X: dict):
    """
    Predicts whether a company will go bankrupt or not based on its financial features.
    
    Parameters:
    X (dict): A dictionary containing the financial features of the company.
              The keys should be 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'.
    
    Returns:
    prediction (int): 0 if the company will not go bankrupt, 1 if it will.
    nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """
    # Initialize the nodes list with None values
    nodes = 2 * [None]
    
    # First node: Working Capital/Total Assets <= 0.20
    nodes[0] = X['WC/TA'] <= 0.20
    
    # If the first node is true, use EBIT/Total Assets as the second feature
    if nodes[0]:
        # Second node: Earnings Before Interest and Tax/Total Assets <= 0.10
        nodes[1] = X['EBIT/TA'] <= 0.10
        
        # Prediction based on the first two nodes
        if nodes[1]:
            prediction = 1  # High risk of bankruptcy
        else:
            prediction = 0  # At risk but not as high
    else:
        # If the first node is false, use Book Value of Equity/Book Value of Liabilities as the second feature
        nodes[1] = X['BVE/BVL'] >= 2.0
        
        # Prediction based on the first two nodes
        if nodes[1]:
            prediction = 0  # Low risk of bankruptcy
        else:
            prediction = 0  # Monitor but low risk
    
    # Convert boolean values to integers (1 for True, 0 for False)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes

# Example usage:
X = {
