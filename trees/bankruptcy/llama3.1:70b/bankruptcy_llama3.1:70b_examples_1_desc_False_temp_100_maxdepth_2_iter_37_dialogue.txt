```python
def predict(X: dict):
    nodes = 2 * [None]
    
    # First node checks if Retained Earnings/Total Assets <= -0.04
    nodes[0] = X["RE/TA"] <= -0.04
    
    # Second node checks if Working Capital/Total Assets <= 0.23 (only evaluated if first node is False)
    nodes[1] = not nodes[0] and X["WC/TA"] <= 0.23
    
    # Use the results of the nodes to make a prediction
    if nodes[0]:
        prediction = 1  # Bankrupt
    elif nodes[1]:
        prediction = 1  # Bankrupt
    else:
        prediction = 0  # Not bankrupt

    return prediction, [int(node) for node in nodes]
```
