```python
def predict(X: dict):
    nodes = [None] * 2
    nodes[0] = X["RE/TA"] <= -0.20
    nodes[1] = X["WC/TA"] <= 0.30

    if nodes[0]:

    else:

        else:
            prediction = 0

    return prediction, [int(node) for node in nodes]
```
