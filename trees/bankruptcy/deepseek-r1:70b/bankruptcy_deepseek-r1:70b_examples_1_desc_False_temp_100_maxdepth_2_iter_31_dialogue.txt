def predict(X):

    bve_bvl = X['BVE/BVL']

    # Initialize nodes to 0 (conditions not met)
    node0 = 0
    node1 = 0

    if ebit_ta <= 0.10:
        node0 = 1
        if bve_bvl <= 1.00:
