```python
def predict(X):
    nodes = [None, None]
    
    # First split: EBIT/TA <= 0.15
    ebit_ta = X['EBIT/TA']
    nodes[0] = ebit_ta <= 0.15
    
    if nodes[0]:
        prediction = 1  # Bankrupt
    else:
        # Second split: S/TA <= 0.80
        s_ta = X['S/TA']
        nodes[1] = s_ta <= 0.80
        
        if nodes[1]:
            prediction = 0  # Not bankrupt
        else:
            prediction = 0  # Not bankrupt
    
    return prediction, nodes
```
