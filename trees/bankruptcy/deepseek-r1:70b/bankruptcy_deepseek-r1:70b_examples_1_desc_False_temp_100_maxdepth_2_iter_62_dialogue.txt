def predict(X):
    nodes = [None, None]
    
    # First node: EBIT/TA <= 0.15
    ebit_condition = X['EBIT/TA'] <= 0.15
    nodes[0] = int(ebit_condition)
    
    if ebit_condition:
        # Second node: BVE/BVL > 2
        bve_bvl_condition = X['BVE/BVL'] > 2
        nodes[1] = int(bve_bvl_condition)
        
        prediction = 0 if bve_bvl_condition else 1
    else:
        # Second node: S/TA > 1
        sales_ta_condition = X['S/TA'] > 1
        nodes[1] = int(sales_ta_condition)
        
        prediction = 0 if sales_ta_condition else 1
    
    return prediction, nodes
