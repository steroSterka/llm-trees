        prediction = 1  # Company is likely bankrupt if EBIT/TA is below threshold
    else:

        nodes[1] = X["BVE/BVL"] <= bve_bvl_threshold
        
        if nodes[1]:

        else:
            prediction = 0  # Company is likely not bankrupt
    
    return prediction, nodes

# Example usage:
