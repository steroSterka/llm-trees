def predict(X):
    # Define the threshold values
    threshold_ebit = 0.10
    threshold_wc = 0.20
    threshold_bve = 1.00

    nodes = [None, None]

    # Compute first node condition: EBIT/TA <= threshold_ebit
    ebit_condition = X['EBIT/TA'] <= threshold_ebit
    nodes[0] = ebit_condition

    if ebit_condition:
        # Second node is WC/TA condition
        wc_condition = X['WC/TA'] <= threshold_wc
        nodes[1] = wc_condition
    else:
        # Second node is BVE/BVL condition
        bve_condition = X['BVE/BVL'] <= threshold_bve
        nodes[1] = bve_condition

    # Determine the prediction based on the conditions
    if ebit_condition:
