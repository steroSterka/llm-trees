```python
def predict(X: dict):
    nodes = [None, None]  # List to store truth values of inner nodes
    
    # First condition: EBIT/TA <= 0.05
    ebit_condition = X["EBIT/TA"] <= 0.05
    nodes[0] = 1 if ebit_condition else 0
    
    prediction = None
    
    if ebit_condition:
        # Leaf node: Bankrupt (target = 1)
        prediction = 1
    else:
        # Second condition: WC/TA <= 0.20
        wc_condition = X["WC/TA"] <= 0.20
        nodes[1] = 1 if wc_condition else 0
        
        if wc_condition:
            # Leaf node: Bankrupt (target = 1)
            prediction = 1
        else:
            # Leaf node: Non-Bankrupt (target = 0)
            prediction = 0
    
    return prediction, nodes
```
