```python
def predict(X: dict):
    """

    """
    # Node 0: WC/TA <= 0.10
    node0_condition = X['WC/TA'] <= 0.10
    
    # Node 1: RE/TA <= 0.25 (only evaluated if WC/TA > 0.10)
    node1_condition = X['RE/TA'] <= 0.25
    
    nodes = [node0_condition, node1_condition]

    if node0_condition:
        prediction = 1  # Bankrupt
    else:
        if node1_condition:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt
            
    return prediction, nodes
```
