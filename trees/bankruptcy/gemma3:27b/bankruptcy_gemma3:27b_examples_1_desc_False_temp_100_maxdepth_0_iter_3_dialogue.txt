```python
def predict(X: dict):
    """
    Predicts company bankruptcy based on financial features using a decision tree.

    Args:
        X (dict): A dictionary containing the financial features of a company.
                  Keys must be 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'.

    Returns:
        tuple: A tuple containing the predicted bankruptcy status (0 or 1) and a list
               representing the truth values of the inner nodes.
    """

    nodes = [None] * 3  # Three inner nodes

    # Node 1: Working Capital/Total Assets <= 0.0
    nodes[0] = X['WC/TA'] <= 0.0

    # Node 2: Retained Earnings/Total Assets <= 0.1 (if WC/TA > 0.0)
    if not nodes[0]:
        nodes[1] = X['RE/TA'] <= 0.1

        nodes[1] = False  # Not reached if WC/TA <= 0.0

    # Node 3: Earnings Before Interest and Tax/Total Assets <= -0.05 (if WC/TA > 0.0 and RE/TA > 0.1)
    if not nodes[0] and not nodes[1]:
        nodes[2] = X['EBIT/TA'] <= -0.05
    else:
        nodes[2] = False  # Not reached if conditions aren't met

    # Prediction
    if nodes[0]:
        prediction = 1  # Bankrupt
    elif nodes[1]:
        prediction = 1  # Bankrupt
    elif nodes[2]:
        prediction = 1  # Bankrupt
    else:
        prediction = 0  # Not Bankrupt

    return prediction, nodes
```
