def predict(X: dict):
    """
    Predicts company bankruptcy based on financial features using a decision tree.

    Args:
        X (dict): A dictionary containing the financial features of the company.
                   Keys should be: 'WC/TA', 'RE/TA', 'EBIT/TA', 'S/TA', 'BVE/BVL'.

    Returns:
        tuple: A tuple containing:
            - prediction (int): The predicted bankruptcy status (0 = no, 1 = yes).
            - nodes (list): A list of boolean values representing the truth values of the inner nodes.
    """

    nodes = []

    # Node 1: WC/TA <= 0.0
    wc_ta_condition = X['WC/TA'] <= 0.0
    nodes.append(wc_ta_condition)

    if wc_ta_condition:
        # Node 2: RE/TA <= 0.1
        re_ta_condition = X['RE/TA'] <= 0.1
        nodes.append(re_ta_condition)

        if re_ta_condition:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt
    else:
        # Node 3: EBIT/TA <= 0.05
        ebit_ta_condition = X['EBIT/TA'] <= 0.05
        nodes.append(ebit_ta_condition)

        if ebit_ta_condition:
            prediction = 1  # Bankrupt
        else:
            prediction = 0  # Not Bankrupt

    return prediction, nodes
