def predict(X: dict):

    Returns:
    prediction (int): 0 if the outcome is bad, 1 if it's good
    nodes (list): A list of boolean values representing the conditions at each node in the decision tree
    """
    # Initialize nodes with None
    nodes = [None] * 7

    nodes[0] = X["wage increase first year"] >= 4.5

    nodes[6] = X["wage increase first year"] > 4.5 and X["working hours"] <= 37
    
    prediction = None
    
    # Start traversing the decision tree from Node 0
    if nodes[0]:

            if nodes[2]:

                prediction = 0  # Bad outcome if working hours are more than 37 and no cost of living adjustment
    else:
        # If wage increase is less than 4.5, check if it's greater than 3.5 (Node 5)
        if nodes[5]:
            # If working hours are less than or equal to 37, it's a good outcome
            if nodes[1]:
                prediction = 1
            else:
                # If working hours are more than 37 and cost of living adjustment is not 'none', it's still a good outcome
                if nodes[2]:
                    prediction = 1
                else:
                    prediction = 0  # Bad outcome if working hours are more than 37 and no cost of living adjustment

    # Convert boolean nodes to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]
    
    return prediction, nodes
