def predict_labor_outcome(X: dict):
    """
    Predicts the outcome of labor negotiations based on a decision tree.
    
    Parameters:
    X (dict): Dictionary containing the input features.
        - 'wage increase first year': wage increase in first year of contract [2.0 .. 7.0]
        - 'working hours': number of working hours during week [35 .. 40]
        - 'pension': employer contributions to pension plan ['none', 'ret_allw', 'empl_contr']
        - 'education allowance': education allowance [True, False]
        - 'contribution to health plan': employer's contribution towards the health plan ['none', 'half', 'full']

    Returns:
    prediction (int): 0 for bad outcome, 1 for good outcome
    nodes (list): List of truth values for inner nodes in the decision tree
    """

    # Initialize nodes list with None values
    nodes = [None] * 6

    # Node 0: Wage increase in first year <= 4.5
    nodes[0] = X['wage increase first year'] <= 4.5
    
    # Node 1: Working hours > 38
    nodes[1] = X['working hours'] > 38
    
    # Node 2: Pension contributions are not 'none'
    nodes[2] = X['pension'] != 'none'
    
    # Node 3: Education allowance is True
    nodes[3] = X['education allowance']
    
    # Node 4: Contribution to health plan is 'full'
    nodes[4] = X['contribution to health plan'] == 'full'

    # Node 5: Wage increase in first year > 4.5 and education allowance is True
    nodes[5] = X['wage increase first year'] > 4.5 and X['education allowance']

    if nodes[0]:  # Wage increase in first year <= 4.5
        if not nodes[2]:  # Pension contributions are 'none'
            prediction = 0  # Bad outcome
        else:
            if nodes[1]:  # Working hours > 38
                prediction = 0  # Bad outcome
            else:
                prediction = 1  # Good outcome
    else:  # Wage increase in first year > 4.5
        if nodes[3] and nodes[4]:  # Education allowance is True and health plan contribution is 'full'
            prediction = 1  # Good outcome
        elif nodes[5]:  # Wage increase in first year > 4.5 and education allowance is True
            prediction = 1  # Good outcome
        else:
            prediction = 0  # Bad outcome

    return prediction, [int(node) for node in nodes]
