def predict(X: dict):
    """
    This function predicts whether a labor negotiation outcome is good or bad based on given features.

    Parameters:
    X (dict): A dictionary containing the features of the labor negotiation.
        - duration: duration of agreement [1..7]
        - wage increase first year: wage increase in first year of contract [2.0 .. 7.0]
        - wage increase second year: age increase in second year of contract [2.0 .. 7.0]
        - wage increase third year: wage increase in third year of contract [2.0 .. 7.0]
        - cost of living adjustment: cost of living allowance [none, tcf, tc]
        - working hours: number of working hours during week [35 .. 40]
        - pension: employer contributions to pension plan [none, ret_allw, empl_contr]
        - standby pay: standby pay [2 .. 25]
        - shift differential: supplement for work on II and III shift [1 .. 25]
        - education allowance: education allowance [true false]
        - statutory holidays: number of statutory holidays [9 .. 15]
        - vacation: number of paid vacation days [ba, avg, gnr]
        - longterm disability assistance: employer's help during employee longterm disabil ity [true , false]
        - contribution to dental plan: employers contribution towards the dental plan [none, half, full]
        - bereavement assistance: employer's financial contribution towards the covering the costs of bereavement [true , false]
        - contribution to health plan: employer's contribution towards the health plan [none, half, full]

    Returns:
    prediction (int): 1 if the labor negotiation outcome is good, 0 otherwise.
    nodes (list): A list representing the truth values of the inner nodes in the decision tree.
    """
    # Initialize nodes
    nodes = 6 * [None]

    # Check wage increase first year
    nodes[0] = X["wage increase first year"] >= 5

    # Check working hours if wage increase is high
    if nodes[0]:
        nodes[1] = X["working hours"] <= 37
    else:
        nodes[1] = False  # This node will not be used in this case, but it's initialized for consistency

    # Check pension contribution
    nodes[2] = X["pension"] == "ret_allw" or X["pension"] == "empl_contr"

    # Check standby pay if wage increase is low and pension contribution is low
    if not nodes[0]:
        nodes[3] = X["standby pay"] > 10
    else:
        nodes[3] = True  # This node will not be used in this case, but it's initialized for consistency

    # Check vacation days if all other conditions are met
    nodes[4] = X["vacation"] == "gnr"

    # Check longterm disability assistance
    nodes[5] = X["longterm disability assistance"] == True

    # Make prediction based on the decision tree
    if nodes[0]:
        if nodes[1]:
            if nodes[2]:
                prediction = 1
            else:
                if nodes[3]:
                    prediction = 1
                else:
                    prediction = 0
        else:
            if nodes[4] and nodes[5]:
                prediction = 1
            else:
                prediction = 0
    else:
        if nodes[2]:
            prediction = 1
        elif nodes[3]:
            if nodes[4] and nodes[5]:
                prediction = 1
            else:
                prediction = 0
        else:
            prediction = 0

    # Convert boolean values to integers (True -> 1, False -> 0)
    nodes = [int(node) for node in nodes]

    return prediction, nodes
